```ts
function readFileAsDataUrl(file: File): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();

    reader.onload = () => {
      if (typeof reader.result !== 'string') {
```

```ts
async function drawImageBox(
  context: DrawContext,
  element: WriterElementV2,
  rect: PdfRect,
): Promise<void> {
  drawBoxBackground(context.page, element, rect);

  const assetId = element.imageAssetId ?? element.assetIds?.[0];

  if (!assetId) return;

  const asset = context.assetsById.get(assetId);

  if (!asset || !asset.dataUrl.startsWith('data:image/')) {
    return;
  }

  const imageBytes = dataUrlToBytes(asset.dataUrl);
  const image = asset.mimeType.includes('png') || asset.dataUrl.startsWith('data:image/png')
    ? await context.doc.embedPng(imageBytes)
    : await context.doc.embedJpg(imageBytes);

  const padding = element.style.padding ?? 0;
  const maxWidth = Math.max(1, rect.width - padding * 2);
  const maxHeight = Math.max(1, rect.height - padding * 2);
  const scale = Math.min(maxWidth / image.width, maxHeight / image.height, 1);
  const width = image.width * scale;
  const height = image.height * scale;

  context.page.drawImage(image, {
    x: rect.x + padding + (maxWidth - width) / 2,
    y: rect.y + padding + (maxHeight - height) / 2,
    width,
    height,
    opacity: clamp(element.style.opacity ?? 1, 0, 1),
  });

  context.embeddedAssetIds.add(asset.id);
}

async function drawTableBox(
  context: DrawContext,
  element: WriterElementV2,
  rect: PdfRect,
): Promise<void> {
  drawBoxBackground(context.page, element, rect);

  const tableData = element.tableData;

  if (!tableData || tableData.columns.length === 0) {
    return;
  }

  const font = await getFontForStyle(context, {
    fontFamily: element.style.fontFamily,
    fontSize: element.style.fontSize,
  });

  const fontBold = await getFontForStyle(context, {
    fontFamily: element.style.fontFamily,
    fontSize: element.style.fontSize,
    bold: true,
  });

  const columns = tableData.columns;
  const headerHeight = 20;
  const rowCount = Math.max(1, tableData.rows.length);
  const rowHeight = Math.max(16, (rect.height - headerHeight) / rowCount);
  const colWidth = rect.width / columns.length;
  const textSize = element.style.fontSize ?? 9;
  const border = parseCssColor(element.style.borderColor, {
    r: 0.58,
    g: 0.64,
    b: 0.72,
  });
  const textColor = parseCssColor(element.style.textColor, {
    r: 0.07,
    g: 0.09,
    b: 0.15,
  });

  for (let colIndex = 0; colIndex < columns.length; colIndex += 1) {
    const column = columns[colIndex];
    const x = rect.x + colIndex * colWidth;
    const y = rect.y + rect.height - headerHeight;

    context.page.drawRectangle({
      x,
      y,
      width: colWidth,
      height: headerHeight,
      color: rgb(0.95, 0.97, 0.99),
      borderColor: rgb(border.r, border.g, border.b),
      borderWidth: 0.5,
    });

    context.page.drawText(column.label ?? '', {
      x: x + 4,
      y: y + 6,
      size: textSize,
      font: fontBold,
      color: rgb(textColor.r, textColor.g, textColor.b),
      maxWidth: Math.max(1, colWidth - 8),
    });
  }

  for (let rowIndex = 0; rowIndex < tableData.rows.length; rowIndex += 1) {
    const row = tableData.rows[rowIndex];
    const y = rect.y + rect.height - headerHeight - (rowIndex + 1) * rowHeight;

    for (let colIndex = 0; colIndex < columns.length; colIndex += 1) {
      const x = rect.x + colIndex * colWidth;
      const cell = row.cells[colIndex];

      context.page.drawRectangle({
        x,
        y,
        width: colWidth,
        height: rowHeight,
        borderColor: rgb(border.r, border.g, border.b),
        borderWidth: 0.5,
      });

      if (cell?.text) {
        context.page.drawText(cell.text, {
          x: x + 4,
          y: y + Math.max(4, rowHeight - textSize - 4),
          size: textSize,
          font,
          color: rgb(textColor.r, textColor.g, textColor.b),
          maxWidth: Math.max(1, colWidth - 8),
        });
      }
    }
  }
}
```

```ts
export function distributeSelectedWriterElements(
  mode: WriterDistributeMode,
): WriterSelectionActionResult {
  const selected = getEditableSelection();

  if (selected.length < 3) {
    return {
      changed: false,
      ids: selected.map((element) => element.id),
      message: 'Select at least three editable Writer V2 elements to distribute.',
    };
  }

  const sorted = [...selected].sort((a, b) =>
    mode === 'horizontal'
      ? a.rect.x - b.rect.x
      : a.rect.y - b.rect.y,
  );

  const first = sorted[0];
  const last = sorted[sorted.length - 1];

  if (mode === 'horizontal') {
    const start = first.rect.x;
    const end = last.rect.x + last.rect.width;
    const totalWidth = sorted.reduce((sum, element) => sum + element.rect.width, 0);
    const gap = (end - start - totalWidth) / (sorted.length - 1);

    let cursor = start;
    const updates = sorted.map((element) => {
      const update = patchRect(element, { x: cursor });
      cursor += element.rect.width + gap;
      return update;
    });

    return commitDistribution(sorted, updates, mode);
  }

  const start = first.rect.y;
  const end = last.rect.y + last.rect.height;
  const totalHeight = sorted.reduce((sum, element) => sum + element.rect.height, 0);
  const gap = (end - start - totalHeight) / (sorted.length - 1);

  let cursor = start;
  const updates = sorted.map((element) => {
    const update = patchRect(element, { y: cursor });
    cursor += element.rect.height + gap;
    return update;
  });

  return commitDistribution(sorted, updates, mode);
}

  store.dispatchWriterCommand({
    type: 'BATCH_UPDATE_WRITER_ELEMENTS',

  });

    changed: true,
    ids: selected.map((element) => element.id),
    message: `Distributed ${selected.length} Writer V2 element(s) ${mode}.`,
  };

function getEditableSelection(): WriterElementV2[] {
  const store = useWriterV2Store.getState();
  const ids = new Set(store.selectedIds);

  return store.elements.filter((element) =>
    ids.has(element.id) &&
    !element.locked &&
    !element.hidden,
  );
}

function selectionBounds(elements: WriterElementV2[]): PdfRect {
  const minX = Math.min(...elements.map((element) => element.rect.x));
  const minY = Math.min(...elements.map((element) => element.rect.y));
  const maxX = Math.max(...elements.map((element) => element.rect.x + element.rect.width));
  const maxY = Math.max(...elements.map((element) => element.rect.y + element.rect.height));

  return {
    x: minX,
    y: minY,
    width: maxX - minX,
    height: maxY - minY,
  };
}

function patchRect(
  element: WriterElementV2,
  rectPatch: Partial<PdfRect>,
): {
  id: string;
  patch: Partial<WriterElementV2>;
} {
  return {
    id: element.id,
    patch: {
      rect: {
        ...element.rect,
        ...rectPatch,
      },
    },
  };
}

function cloneWriterElementForDuplicate(
  element: WriterElementV2,
  now: string,
  zIndex: number,
): WriterElementV2 {
  return {
    ...element,
    id: createWriterElementId(),
    rect: {
      ...element.rect,
      x: element.rect.x + DUPLICATE_OFFSET,
      y: Math.max(0, element.rect.y - DUPLICATE_OFFSET),
    },
    assetIds: element.assetIds ? [...element.assetIds] : undefined,
    tableData: element.tableData
      ? {
          columns: element.tableData.columns.map((column) => ({ ...column })),
          rows: element.tableData.rows.map((row) => ({
            ...row,
            cells: row.cells.map((cell) => ({ ...cell })),
          })),
        }
      : undefined,
    callout: element.callout
      ? {
          ...element.callout,
          anchorPoint: element.callout.anchorPoint
            ? {
                x: element.callout.anchorPoint.x + DUPLICATE_OFFSET,
                y: Math.max(0, element.callout.anchorPoint.y - DUPLICATE_OFFSET),
              }
            : undefined,
        }
      : undefined,
    style: {
      ...element.style,
      zIndex,
    },
    createdAt: now,
    updatedAt: now,
    locked: false,
    hidden: false,
  };
}

function createWriterElementId(): string {
  if (globalThis.crypto?.randomUUID) {
    return `writer-v2-${globalThis.crypto.randomUUID()}`;
  }

  return `writer-v2-${Date.now().toString(36)}-${Math.random()
    .toString(36)
    .slice(2)}`;
}
```

```tsx
  duplicateSelectedWriterElements,
  nudgeSelectedWriterElements,
} from '@/core/writer-v2/writerSelectionActions';

/**
 * Document-level Writer V2 keyboard shortcuts.
 * This component renders nothing.
 */
export const WriterKeyboardHandler: React.FC = () => {
  React.useEffect(() => {
    const handler = (event: KeyboardEvent) => {
      const target = event.target as HTMLElement | null;

      if (
        target instanceof HTMLInputElement ||
        target instanceof HTMLTextAreaElement ||
        target?.isContentEditable
      ) {
        return;
      }

      const store = useWriterV2Store.getState();

      if (store.editingId !== null) {
        return;
      }

      const hasSelection = store.selectedIds.length > 0;
      const isMac = navigator.platform.toUpperCase().includes('MAC');
      const ctrl = isMac ? event.metaKey : event.ctrlKey;
      const step = event.shiftKey ? 10 : 1;

      if (hasSelection && isArrowKey(event.key)) {
        const delta = nudgeDelta(event.key, step);

        if (delta) {
          event.preventDefault();
          nudgeSelectedWriterElements(delta.dx, delta.dy);
          return;
        }
      }

      if ((event.key === 'Delete' || event.key === 'Backspace') && hasSelection) {
        event.preventDefault();
        deleteSelectedWriterElements();
        return;
      }

      if (ctrl && event.key.toLowerCase() === 'd' && hasSelection) {
        event.preventDefault();
        duplicateSelectedWriterElements();
        return;
      }

      if (event.key === 'Escape' && hasSelection) {
        event.preventDefault();
        store.dispatchWriterCommand({
          type: 'CLEAR_WRITER_SELECTION',
        });
        return;
      }

      if (ctrl && event.key.toLowerCase() === 'z' && !event.shiftKey) {
        event.preventDefault();
        store.undoWriter();
        return;
      }

      if (ctrl && event.key.toLowerCase() === 'z' && event.shiftKey) {
        event.preventDefault();
        store.redoWriter();
      }
    };

    document.addEventListener('keydown', handler);
    return () => document.removeEventListener('keydown', handler);
  }, []);

  return null;
};

function isArrowKey(key: string): boolean {
  return key === 'ArrowLeft' ||
    key === 'ArrowRight' ||
    key === 'ArrowUp' ||
    key === 'ArrowDown';
}

/**
 * Writer rects use PDF coordinates, origin bottom-left.
 */
function nudgeDelta(
  key: string,
  step: number,
): { dx: number; dy: number } | null {
  switch (key) {
    case 'ArrowLeft':
      return { dx: -step, dy: 0 };

    case 'ArrowRight':
      return { dx: step, dy: 0 };

    case 'ArrowUp':
      return { dx: 0, dy: step };

    case 'ArrowDown':
      return { dx: 0, dy: -step };

    default:
      return null;
  }
}
```

```ts
describe('writer selection object controls', () => {
  beforeEach(() => {
    useWriterV2Store.getState().clearWriterV2();
  });

  it('nudges selected tableBox and imageBox objects as one undoable batch', () => {
    useWriterV2Store.getState().hydrateWriterSession({

        element('table-1', {
          type: 'tableBox',
          rect: { x: 10, y: 20, width: 100, height: 80 },
        }),
        element('image-1', {
          type: 'imageBox',
          rect: { x: 200, y: 300, width: 120, height: 90 },
        }),
      ],
    });

    useWriterV2Store.getState().dispatchWriterCommand({
      type: 'SET_WRITER_SELECTION',
      ids: ['table-1', 'image-1'],
    });

    nudgeSelectedWriterElements(5, -3);

    const state = useWriterV2Store.getState();

    expect(state.elements.find((item) => item.id === 'table-1')?.rect).toMatchObject({
      x: 15,
      y: 17,
```

```ts
import type {
  WriterElementV2,
  WriterTableCell,
  WriterTableData,
  WriterTableRow,
} from './types';
import { useWriterV2Store } from './writerV2Store';

export interface WriterTableActionResult {
  changed: boolean;
  message: string;
}

export function updateActiveTableCell(
  rowId: string,
  cellId: string,
  text: string,
): WriterTableActionResult {
  const table = getActiveTableBox();

  if (!table?.tableData) {
    return {
      changed: false,
      message: 'No active Writer table selected.',
    };
  }

  const tableData = cloneTableData(table.tableData);

  for (const row of tableData.rows) {
    if (row.id !== rowId) continue;

    for (const cell of row.cells) {
      if (cell.id === cellId) {
        cell.text = text;
      }
    }
  }

  return commitTableData(table, tableData, 'Updated Writer table cell.');
}

export function addActiveTableRow(): WriterTableActionResult {
  const table = getActiveTableBox();

  if (!table?.tableData) {
    return {
      changed: false,
      message: 'No active Writer table selected.',
    };
  }

  const tableData = cloneTableData(table.tableData);

  tableData.rows.push({
    id: createTableId('row'),
    cells: tableData.columns.map(() => ({
      id: createTableId('cell'),
      text: '',
    })),
  });

  return commitTableData(table, tableData, 'Added Writer table row.');
}

export function removeActiveTableRow(): WriterTableActionResult {
  const table = getActiveTableBox();

  if (!table?.tableData) {
    return {
      changed: false,
      message: 'No active Writer table selected.',
    };
  }

  if (table.tableData.rows.length <= 1) {
    return {
      changed: false,
      message: 'Writer table must keep at least one row.',
    };
  }

  const tableData = cloneTableData(table.tableData);
  tableData.rows.pop();

  return commitTableData(table, tableData, 'Removed Writer table row.');
}

export function addActiveTableColumn(): WriterTableActionResult {
  const table = getActiveTableBox();

  if (!table?.tableData) {
    return {
      changed: false,
      message: 'No active Writer table selected.',
    };
  }

  const tableData = cloneTableData(table.tableData);
  const nextIndex = tableData.columns.length + 1;

  tableData.columns.push({
    id: createTableId('col'),
    label: `Column ${nextIndex}`,
    width: 1,
  });

  tableData.rows = tableData.rows.map((row) => ({
    ...row,
    cells: [
      ...row.cells,
      {
        id: createTableId('cell'),
        text: '',
      },
    ],
  }));

  return commitTableData(table, tableData, 'Added Writer table column.');
}

export function removeActiveTableColumn(): WriterTableActionResult {
  const table = getActiveTableBox();

  if (!table?.tableData) {
    return {
      changed: false,
      message: 'No active Writer table selected.',
    };
  }

  if (table.tableData.columns.length <= 1) {
    return {
      changed: false,
      message: 'Writer table must keep at least one column.',
    };
  }

  const tableData = cloneTableData(table.tableData);

  tableData.columns.pop();
  tableData.rows = tableData.rows.map((row) => ({
    ...row,
    cells: row.cells.slice(0, Math.max(1, row.cells.length - 1)),
  }));

  return commitTableData(table, tableData, 'Removed Writer table column.');
}

export function updateActiveTableColumnLabel(
  columnId: string,
  label: string,
): WriterTableActionResult {
  const table = getActiveTableBox();

  if (!table?.tableData) {
    return {
      changed: false,
      message: 'No active Writer table selected.',
    };
  }

  const tableData = cloneTableData(table.tableData);

  tableData.columns = tableData.columns.map((column) =>
    column.id === columnId
      ? {
          ...column,
          label,
        }
      : column,
  );

  return commitTableData(table, tableData, 'Updated Writer table column label.');
}

export function getActiveTableBox(): WriterElementV2 | null {
  const store = useWriterV2Store.getState();
  const activeId = store.activeId ?? store.selectedIds.at(-1);

  if (!activeId) return null;

  const element = store.elements.find((item) => item.id === activeId);

  if (!element || element.type !== 'tableBox' || element.locked || element.hidden) {
    return null;
  }

  return element;
}

function commitTableData(
  table: WriterElementV2,
  tableData: WriterTableData,
  message: string,
): WriterTableActionResult {
  const store = useWriterV2Store.getState();

  const result = store.dispatchWriterCommand({
    type: 'UPDATE_WRITER_ELEMENT',
    id: table.id,
    patch: {
      tableData,
      plainText: summarizeTable(tableData),
    },
  });

  return {
    changed: result.changed,
    message,
  };
}

function summarizeTable(tableData: WriterTableData): string {
  return `Table ${tableData.columns.length}x${tableData.rows.length}`;
}

function cloneTableData(tableData: WriterTableData): WriterTableData {
  return {
    columns: tableData.columns.map((column) => ({ ...column })),
    rows: tableData.rows.map((row): WriterTableRow => ({
      ...row,
      cells: row.cells.map((cell): WriterTableCell => ({ ...cell })),
    })),
  };
}

function createTableId(prefix: string): string {
  if (globalThis.crypto?.randomUUID) {
    return `${prefix}-${globalThis.crypto.randomUUID()}`;
  }

  return `${prefix}-${Date.now().toString(36)}-${Math.random()
    .toString(36)
    .slice(2)}`;
}
```

```ts
import type { WriterAssetRecordV2 } from './persistence/indexedDbWriterStore';
import type { WriterElementV2 } from './types';
import { useWriterV2Store } from './writerV2Store';

export interface WriterImageReplaceResult {
  changed: boolean;
  message: string;
}

export async function replaceActiveImageBoxAsset(
  file: File,
): Promise<WriterImageReplaceResult> {
  const store = useWriterV2Store.getState();
  const activeId = store.activeId ?? store.selectedIds.at(-1);

  if (!activeId) {
    return {
      changed: false,
      message: 'No active Writer image selected.',
    };
  }

  const element = store.elements.find((item) => item.id === activeId);

  if (!element || element.type !== 'imageBox' || element.locked || element.hidden) {
    return {
      changed: false,
      message: 'No editable Writer image selected.',
    };
  }

  if (!/^image\/(png|jpeg|webp)$/i.test(file.type)) {
    return {
      changed: false,
      message: 'Only PNG, JPG, and WEBP images are supported.',
    };
  }

  const dataUrl = await readFileAsDataUrl(file);
  const size = await measureImageDataUrl(dataUrl);
  const now = new Date().toISOString();
  const assetId = createWriterAssetId();

  const asset: WriterAssetRecordV2 = {
    id: assetId,
    documentFingerprint:
      store.currentDocumentFingerprint ??
      element.documentFingerprint,
    kind: 'image',
    mimeType: file.type,
    dataUrl,
    width: size.width,
    height: size.height,
    createdAt: now,
    updatedAt: now,
  };

  store.addWriterAssets([asset]);

  const result = store.dispatchWriterCommand({
    type: 'UPDATE_WRITER_ELEMENT',
    id: element.id,
    patch: {
      plainText: file.name,
      imageAssetId: assetId,
      assetIds: [assetId],
    },
  });

  return {
    changed: result.changed,
    message: result.changed
      ? 'Replaced Writer image asset.'
      : result.reason,
  };
}

export function clearActiveImageBoxAsset(): WriterImageReplaceResult {
  const store = useWriterV2Store.getState();
  const activeId = store.activeId ?? store.selectedIds.at(-1);

  if (!activeId) {
    return {
      changed: false,
      message: 'No active Writer image selected.',
    };
  }

  const element = store.elements.find((item) => item.id === activeId);

  if (!element || element.type !== 'imageBox' || element.locked || element.hidden) {
    return {
      changed: false,
      message: 'No editable Writer image selected.',
    };
  }

  const result = store.dispatchWriterCommand({
    type: 'UPDATE_WRITER_ELEMENT',
    id: element.id,
    patch: {
      plainText: 'Image',
      imageAssetId: undefined,
      assetIds: [],
    },
  });

  return {
    changed: result.changed,
    message: result.changed
      ? 'Cleared Writer image asset.'
      : result.reason,
  };
}

export function getActiveImageBox(): WriterElementV2 | null {
  const store = useWriterV2Store.getState();
  const activeId = store.activeId ?? store.selectedIds.at(-1);

  if (!activeId) return null;

  const element = store.elements.find((item) => item.id === activeId);

  if (!element || element.type !== 'imageBox' || element.locked || element.hidden) {
    return null;
  }

  return element;
}

function readFileAsDataUrl(file: File): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();

    reader.onload = () => {
      if (typeof reader.result !== 'string') {
        reject(new Error('Image import did not produce a data URL.'));
        return;
      }

      resolve(reader.result);
    };

    reader.onerror = () => reject(reader.error ?? new Error('Failed to read image file.'));
    reader.readAsDataURL(file);
  });
}

function measureImageDataUrl(
  dataUrl: string,
): Promise<{ width: number; height: number }> {
  return new Promise((resolve, reject) => {
    const image = new Image();

    image.onload = () => {
      resolve({
        width: image.naturalWidth || image.width || 1,
        height: image.naturalHeight || image.height || 1,
      });
    };

    image.onerror = () => reject(new Error('Failed to measure imported image.'));
    image.src = dataUrl;
  });
}

function createWriterAssetId(): string {
  if (globalThis.crypto?.randomUUID) {
    return `writer-asset-${globalThis.crypto.randomUUID()}`;
  }

  return `writer-asset-${Date.now().toString(36)}-${Math.random()
    .toString(36)
    .slice(2)}`;
}
```

```ts
  assetCount: number;
  issues: WriterExportPreflightIssue[];
  byPage: Array<{
    pageIndex: number;
    visibleElementCount: number;
    hiddenElementCount: number;
  }>;
}

const TEXT_TYPES = new Set<WriterElementV2['type']>([
  'richTextBox',
  'textCallout',
  'textStamp',
  'reviewLabel',
  'richNote',
]);

export async function runCurrentWriterExportPreflight(): Promise<WriterExportPreflightResult> {
  const session = useSessionStore.getState();
  const writer = useWriterV2Store.getState();

  return runWriterExportPreflight({
    sourceBytes: session.workingBytes ?? session.originalBytes,
    pageCount: session.pageCount,
    elements: writer.elements,
    assets: writer.assetRecords,
  });
}

export async function runWriterExportPreflight(
  input: WriterExportPreflightInput,
): Promise<WriterExportPreflightResult> {
  const issues: WriterExportPreflightIssue[] = [];

  if (!input.sourceBytes || input.pageCount <= 0) {
    issues.push({
      code: 'NO_PDF_OPEN',
      severity: 'error',
      message: 'Open a PDF before exporting Writer content.',
    });
  }

  const pageSizes = input.sourceBytes
    ? await safeReadPageSizes(input.sourceBytes)
    : [];

  const assetsById = new Map(input.assets.map((asset) => [asset.id, asset]));
  const hiddenElements = input.elements.filter((element) => element.hidden);
  const visibleElements = input.elements.filter((element) => !element.hidden);

  if (input.elements.length === 0) {
    issues.push({
      code: 'NO_WRITER_OBJECTS',
      severity: 'info',
      message: 'No Writer V2 objects exist in this document.',
    });
  }

  if (input.elements.length > 0 && visibleElements.length === 0) {
    issues.push({
      code: 'NO_VISIBLE_WRITER_OBJECTS',
      severity: 'warning',
      message: 'All Writer V2 objects are hidden; flattened export will not show Writer content.',
    });
  }

  if (hiddenElements.length > 0) {
    issues.push({
      code: 'HIDDEN_OBJECTS_EXCLUDED',
      severity: 'info',
      message: `${hiddenElements.length} hidden Writer object(s) will be excluded from flattened export.`,
    });
  }

  for (const element of input.elements) {
    collectElementIssues({
      element,
      assetsById,
      pageCount: input.pageCount,
      pageSize: pageSizes.find((page) => page.pageIndex === element.pageIndex),
      issues,
    });
  }

  const byPage = Array.from({ length: Math.max(0, input.pageCount) }, (_, pageIndex) => {
    const pageElements = input.elements.filter((element) => element.pageIndex === pageIndex);

    return {
      pageIndex,
      visibleElementCount: pageElements.filter((element) => !element.hidden).length,
      hiddenElementCount: pageElements.filter((element) => element.hidden).length,
    };
  });

  const hasErrors = issues.some((issue) => issue.severity === 'error');
  const hasWarnings = issues.some((issue) => issue.severity === 'warning');

  return {
    ok: !hasErrors,
    hasErrors,
    hasWarnings,
    pageCount: input.pageCount,
    totalElementCount: input.elements.length,
    visibleElementCount: visibleElements.length,
    hiddenElementCount: hiddenElements.length,
    textElementCount: input.elements.filter((element) => TEXT_TYPES.has(element.type)).length,
    tableElementCount: input.elements.filter((element) => element.type === 'tableBox').length,
    imageElementCount: input.elements.filter((element) => element.type === 'imageBox').length,
    assetCount: input.assets.length,
    issues,
    byPage,
  };
}

function collectElementIssues({
  element,
  assetsById,
  pageCount,
  pageSize,
  issues,
}: {
  element: WriterElementV2;
  assetsById: Map<string, WriterAssetRecordV2>;
  pageCount: number;
  pageSize?: WriterExportPreflightPageSize;
  issues: WriterExportPreflightIssue[];
}): void {
  if (element.pageIndex < 0 || element.pageIndex >= pageCount) {
    issues.push({
      code: 'OBJECT_OUTSIDE_PAGE_RANGE',
      severity: 'error',
      elementId: element.id,
      pageIndex: element.pageIndex,
      message: `Writer object ${shortId(element.id)} targets page ${element.pageIndex + 1}, outside document page range.`,
    });
  }

  if (element.rect.width <= 0 || element.rect.height <= 0) {
    issues.push({
      code: 'OBJECT_ZERO_SIZE',
      severity: 'error',
      elementId: element.id,
      pageIndex: element.pageIndex,
      message: `Writer object ${shortId(element.id)} has zero or negative size.`,
    });
  }

  if (pageSize && isOutsidePageBounds(element, pageSize)) {
    issues.push({
      code: 'OBJECT_OUTSIDE_PAGE_BOUNDS',
      severity: 'warning',
      elementId: element.id,
      pageIndex: element.pageIndex,
      message: `Writer object ${shortId(element.id)} is partially outside page ${element.pageIndex + 1}.`,
    });
  }

  if ((element.style.opacity ?? 1) > 0 && (element.style.opacity ?? 1) < 0.05) {
    issues.push({
      code: 'OBJECT_LOW_OPACITY',
      severity: 'warning',
      elementId: element.id,
      pageIndex: element.pageIndex,
      message: `Writer object ${shortId(element.id)} has very low opacity and may be invisible after export.`,
    });
  }

  if (TEXT_TYPES.has(element.type) && !element.plainText?.trim() && !element.editorStateJson?.trim()) {
    issues.push({
      code: 'EMPTY_TEXT_OBJECT',
      severity: 'warning',
      elementId: element.id,
      pageIndex: element.pageIndex,
      message: `Writer text object ${shortId(element.id)} is empty.`,
    });
  }

  if (element.type === 'imageBox') {
    collectImageIssues(element, assetsById, issues);
  }

  if (element.type === 'tableBox') {
    collectTableIssues(element, issues);
  }
}

function collectImageIssues(
  element: WriterElementV2,
  assetsById: Map<string, WriterAssetRecordV2>,
  issues: WriterExportPreflightIssue[],
): void {
  const assetId = element.imageAssetId ?? element.assetIds?.[0];

  if (!assetId) {
    issues.push({
      code: 'EMPTY_IMAGE_OBJECT',
      severity: 'error',
      elementId: element.id,
      pageIndex: element.pageIndex,
      message: `Writer image object ${shortId(element.id)} has no image asset assigned.`,
    });
    return;
  }

  if (!assetsById.has(assetId)) {
    issues.push({
      code: 'MISSING_IMAGE_ASSET',
      severity: 'error',
      elementId: element.id,
      pageIndex: element.pageIndex,
      message: `Writer image object ${shortId(element.id)} references missing asset ${shortId(assetId)}.`,
    });
  }
}

function collectTableIssues(
  element: WriterElementV2,
  issues: WriterExportPreflightIssue[],
): void {
  const table = element.tableData;

  if (!table || table.columns.length === 0 || table.rows.length === 0) {
    issues.push({
      code: 'EMPTY_TABLE_OBJECT',
      severity: 'warning',
      elementId: element.id,
      pageIndex: element.pageIndex,
      message: `Writer table object ${shortId(element.id)} has no table data.`,
    });
    return;
  }

  const expectedCellCount = table.columns.length;
  const mismatchedRow = table.rows.find((row) => row.cells.length !== expectedCellCount);

  if (mismatchedRow) {
    issues.push({
      code: 'TABLE_SHAPE_MISMATCH',
      severity: 'error',
      elementId: element.id,
      pageIndex: element.pageIndex,
      message: `Writer table object ${shortId(element.id)} has a row/cell count mismatch.`,
    });
  }
}

function isOutsidePageBounds(
  element: WriterElementV2,
  pageSize: WriterExportPreflightPageSize,
): boolean {
  return element.rect.x < 0 ||
    element.rect.y < 0 ||
    element.rect.x + element.rect.width > pageSize.width ||
    element.rect.y + element.rect.height > pageSize.height;
}

async function safeReadPageSizes(
  bytes: Uint8Array,
): Promise<WriterExportPreflightPageSize[]> {
  try {
    const { PDFDocument } = await import('pdf-lib');
    const pdf = await PDFDocument.load(bytes);

    return pdf.getPages().map((page, pageIndex) => {
      const size = page.getSize();

      return {
        pageIndex,
        width: size.width,
        height: size.height,
      };
    });
  } catch {
    return [];
  }
}

function shortId(id: string): string {
  return id.length <= 8 ? id : id.slice(0, 8);
}
```

# 1. New file — autosave persistence helper

## `frontend/src/core/writer-v2/persistence/writerAutosave.ts`

```ts
import { useSessionStore } from '@/core/session/store';
import { useWriterV2Store } from '../writerV2Store';
import {
  createBestAvailableWriterPersistence,
  createWriterDocumentRecord,
  type WriterDocumentRecordV2,
  type WriterPersistenceAdapterV2,
} from './indexedDbWriterStore';

export interface WriterAutosaveResult {
  changed: boolean;
  documentFingerprint: string | null;
  persistedAt: string | null;
  message: string;
}

export interface WriterDraftLoadResult {
  found: boolean;
  documentFingerprint: string;
  document: WriterDocumentRecordV2 | null;
  assetCount: number;
}

export interface WriterRecoverResult {
  changed: boolean;
  message: string;
}

export async function autosaveCurrentWriterDraft(
  adapter: WriterPersistenceAdapterV2 = createBestAvailableWriterPersistence(),
  options: {
    force?: boolean;
  } = {},
): Promise<WriterAutosaveResult> {
  const writer = useWriterV2Store.getState();
  const documentFingerprint = resolveCurrentWriterFingerprint();

  if (!documentFingerprint) {
    return {
      changed: false,
      documentFingerprint: null,
      persistedAt: null,
      message: 'No open document available for Writer autosave.',
    };
  }

  if (!options.force && !writer.dirty) {
    return {
      changed: false,
      documentFingerprint,
      persistedAt: writer.lastPersistedAt,
      message: 'Writer autosave skipped because there are no changes.',
    };
  }

  const persistedAt = new Date().toISOString();

  await adapter.saveDocument(
    createWriterDocumentRecord(
      documentFingerprint,
      writer.elements,
      persistedAt,
    ),
  );

  for (const asset of writer.assetRecords) {
    await adapter.saveAsset({
      ...asset,
      documentFingerprint,
      updatedAt: persistedAt,
    });
  }

  useWriterV2Store.getState().markWriterPersisted(persistedAt);

  return {
    changed: true,
    documentFingerprint,
    persistedAt,
    message: `Writer autosaved ${writer.elements.length} object(s).`,
  };
}

export async function loadWriterDraftForCurrentDocument(
  adapter: WriterPersistenceAdapterV2 = createBestAvailableWriterPersistence(),
): Promise<WriterDraftLoadResult | null> {
  const documentFingerprint = resolveCurrentWriterFingerprint();

  if (!documentFingerprint) {
    return null;
  }

  const document = await adapter.loadDocument(documentFingerprint);
  const assets = await adapter.listAssets(documentFingerprint);

  return {
    found: Boolean(document),
    documentFingerprint,
    document,
    assetCount: assets.length,
  };
}

export async function recoverWriterDraftForCurrentDocument(
  adapter: WriterPersistenceAdapterV2 = createBestAvailableWriterPersistence(),
): Promise<WriterRecoverResult> {
  const documentFingerprint = resolveCurrentWriterFingerprint();

  if (!documentFingerprint) {
    return {
      changed: false,
      message: 'No open document available for Writer recovery.',
    };
  }

  const document = await adapter.loadDocument(documentFingerprint);

  if (!document) {
    return {
      changed: false,
      message: 'No Writer draft found for this document.',
    };
  }

  const assets = await adapter.listAssets(documentFingerprint);

  useWriterV2Store.getState().hydrateWriterSession({
    documentFingerprint,
    elements: document.elements,
    assetRecords: assets,
    persistedAt: document.updatedAt,
  });

  return {
    changed: true,
    message: `Recovered ${document.elements.length} Writer object(s).`,
  };
}

export async function discardWriterDraftForCurrentDocument(
  adapter: WriterPersistenceAdapterV2 = createBestAvailableWriterPersistence(),
): Promise<WriterRecoverResult> {
  const documentFingerprint = resolveCurrentWriterFingerprint();

  if (!documentFingerprint) {
    return {
      changed: false,
      message: 'No open document available for Writer draft discard.',
    };
  }

  const assets = await adapter.listAssets(documentFingerprint);

  await adapter.deleteDocument(documentFingerprint);

  for (const asset of assets) {
    await adapter.deleteAsset(asset.id);
  }

  return {
    changed: true,
    message: 'Discarded saved Writer draft for this document.',
  };
}

export function resolveCurrentWriterFingerprint(): string | null {
  const writer = useWriterV2Store.getState();
  const session = useSessionStore.getState();

  return writer.currentDocumentFingerprint ??
    session.documentKey ??
    session.fileName ??
    null;
}
```

```tsx
export const WriterAutosaveController: React.FC = () => {

  const elementCount = useWriterV2Store((state) => state.elements.length);
  const assetCount = useWriterV2Store((state) => state.assetRecords.length);
```

```ts
import type {
  WriterElementV2,
  WriterReviewCategory,
  WriterReviewMetadata,
  WriterReviewPriority,
  WriterReviewStatus,
} from './types';
import { useWriterV2Store } from './writerV2Store';

export interface WriterReviewActionResult {
  changed: boolean;
  message: string;
}

export interface WriterReviewPatch {
  status?: WriterReviewStatus;
  priority?: WriterReviewPriority;
  category?: WriterReviewCategory;
  author?: string;
  assignee?: string;
  dueDate?: string;
}

const REVIEWABLE_TYPES = new Set<WriterElementV2['type']>([
  'richTextBox',
  'textCallout',
  'reviewLabel',
  'richNote',
]);

export function isReviewableWriterElement(element: WriterElementV2): boolean {
  return REVIEWABLE_TYPES.has(element.type);
}

export function ensureActiveWriterReviewMetadata(): WriterReviewActionResult {
  const element = getActiveReviewableElement();

  if (!element) {
    return {
      changed: false,
      message: 'No reviewable Writer object selected.',
    };
  }

  if (element.review) {
    return {
      changed: false,
      message: 'Selected Writer object already has review metadata.',
    };
  }

  const now = new Date().toISOString();

  const review: WriterReviewMetadata = {
    status: 'open',
    priority: 'medium',
    category: defaultCategoryForType(element.type),
    commentId: createReviewId(),
    createdAt: now,
    updatedAt: now,
  };

  const store = useWriterV2Store.getState();

  const result = store.dispatchWriterCommand({
    type: 'UPDATE_WRITER_ELEMENT',
    id: element.id,
    patch: {
      review,
    },
  });

  return {
    changed: result.changed,
    message: result.changed
      ? 'Added review metadata to Writer object.'
      : result.reason,
  };
}

export function updateActiveWriterReviewMetadata(
  patch: WriterReviewPatch,
): WriterReviewActionResult {
  const element = getActiveReviewableElement();

  if (!element) {
    return {
      changed: false,
      message: 'No reviewable Writer object selected.',
    };
  }

  const now = new Date().toISOString();
  const previous = element.review;

  const review: WriterReviewMetadata = {
    status: patch.status ?? previous?.status ?? 'open',
    priority: patch.priority ?? previous?.priority ?? 'medium',
    category: patch.category ?? previous?.category ?? defaultCategoryForType(element.type),
    author: patch.author ?? previous?.author,
    assignee: patch.assignee ?? previous?.assignee,
    dueDate: patch.dueDate ?? previous?.dueDate,
    commentId: previous?.commentId ?? createReviewId(),
    createdAt: previous?.createdAt ?? now,
    updatedAt: now,
  };

  const store = useWriterV2Store.getState();

  const result = store.dispatchWriterCommand({
    type: 'UPDATE_WRITER_ELEMENT',
    id: element.id,
    patch: {
      review,
    },
  });

  return {
    changed: result.changed,
    message: result.changed
      ? 'Updated Writer review metadata.'
      : result.reason,
  };
}

export function clearActiveWriterReviewMetadata(): WriterReviewActionResult {
  const element = getActiveReviewableElement();

  if (!element) {
    return {
      changed: false,
      message: 'No reviewable Writer object selected.',
    };
  }

  const store = useWriterV2Store.getState();

  const result = store.dispatchWriterCommand({
    type: 'UPDATE_WRITER_ELEMENT',
    id: element.id,
    patch: {
      review: undefined,
    },
  });

  return {
    changed: result.changed,
    message: result.changed
      ? 'Cleared Writer review metadata.'
      : result.reason,
  };
}

export function markActiveWriterReviewResolved(): WriterReviewActionResult {
  return updateActiveWriterReviewMetadata({
    status: 'resolved',
  });
}

export function getActiveReviewableElement(): WriterElementV2 | null {
  const store = useWriterV2Store.getState();
  const activeId = store.activeId ?? store.selectedIds.at(-1);

  if (!activeId) return null;

  const element = store.elements.find((item) => item.id === activeId);

  if (!element || element.locked || element.hidden || !isReviewableWriterElement(element)) {
    return null;
  }

  return element;
}

function defaultCategoryForType(
  type: WriterElementV2['type'],
): WriterReviewCategory {
  switch (type) {
    case 'textCallout':
      return 'engineering';

    case 'reviewLabel':
      return 'qa';

    case 'richNote':
      return 'general';

    case 'richTextBox':
      return 'general';

    default:
      return 'general';
  }
}

function createReviewId(): string {
  if (globalThis.crypto?.randomUUID) {
    return `review-${globalThis.crypto.randomUUID()}`;
  }

  return `review-${Date.now().toString(36)}-${Math.random()
    .toString(36)
    .slice(2)}`;
}
```

```ts
  'richTextBox',
  'textCallout',
  'reviewLabel',
  'richNote',
]);

export function collectCurrentWriterCommentSummaryRows(): WriterCommentSummaryRow[] {

  elements: WriterElementV2[],

      status: element.review?.status ?? '',
      priority: element.review?.priority ?? '',
      category: element.review?.category ?? '',
      author: element.review?.author ?? '',
      assignee: element.review?.assignee ?? '',
      dueDate: element.review?.dueDate ?? '',
      createdAt: element.review?.createdAt ?? element.createdAt,
      updatedAt: element.review?.updatedAt ?? element.updatedAt,
    }))
    .sort((a, b) => {
      if (a.page !== b.page) return a.page - b.page;
      return a.type.localeCompare(b.type);
    });
}

export function serializeWriterCommentSummary(
  rows: WriterCommentSummaryRow[],
  format: WriterCommentSummaryFormat,
): string {
  switch (format) {
    case 'csv':
      return serializeCsv(rows);

    case 'markdown':
      return serializeMarkdown(rows);

    case 'json':
      return JSON.stringify(rows, null, 2);
  }
}

export function downloadCurrentWriterCommentSummary(
  format: WriterCommentSummaryFormat,
): {
  changed: boolean;
  message: string;
} {
  const rows = collectCurrentWriterCommentSummaryRows();

  if (rows.length === 0) {
    return {
      changed: false,
      message: 'No visible Writer comments or review objects to export.',
    };
  }

  const session = useSessionStore.getState();
  const baseName = safeBaseName(session.fileName ?? 'writer-comments');
  const extension = format === 'markdown' ? 'md' : format;
  const text = serializeWriterCommentSummary(rows, format);
  const mimeType = mimeTypeForFormat(format);

  downloadTextFile(
    `${baseName}.writer-comments.${extension}`,
    text,
    mimeType,
  );

  return {
    changed: true,
    message: `Exported ${rows.length} Writer comment(s).`,
  };
}

function serializeCsv(rows: WriterCommentSummaryRow[]): string {
  const headers: Array<keyof WriterCommentSummaryRow> = [
    'id',
    'page',
    'type',
    'text',
    'status',
    'priority',
    'category',
    'author',
    'assignee',
    'dueDate',
    'createdAt',
    'updatedAt',
  ];

  const lines = [
    headers.join(','),
    ...rows.map((row) =>
      headers
        .map((header) => csvCell(String(row[header] ?? '')))
        .join(','),
    ),
  ];

  return `${lines.join('\n')}\n`;
}

function serializeMarkdown(rows: WriterCommentSummaryRow[]): string {
  const lines = [
    '# Writer Comment Summary',
    '',
    `Total comments: ${rows.length}`,
    '',
    '| Page | Type | Status | Priority | Category | Assignee | Text |',
    '|---:|---|---|---|---|---|---|',
    ...rows.map((row) =>
      `| ${row.page} | ${escapeMarkdown(row.type)} | ${escapeMarkdown(row.status)} | ${escapeMarkdown(row.priority)} | ${escapeMarkdown(row.category)} | ${escapeMarkdown(row.assignee)} | ${escapeMarkdown(row.text)} |`,
    ),
    '',
  ];

  return lines.join('\n');
}

function csvCell(value: string): string {
  const escaped = value.replaceAll('"', '""');

  return `"${escaped}"`;
}

function escapeMarkdown(value: string): string {
  return value
    .replaceAll('|', '\\|')
    .replaceAll('\n', '<br>');
}

function normalizeText(value: string | undefined): string {
  return (value ?? '')
    .replace(/\s+/g, ' ')
    .trim();
}

function safeBaseName(fileName: string): string {
  return fileName
    .replace(/\.pdfhub\.zip$/i, '')
    .replace(/\.pdf$/i, '')
    .replace(/[^\w.-]+/g, '_')
    .replace(/^_+|_+$/g, '') || 'writer-comments';
}

function mimeTypeForFormat(format: WriterCommentSummaryFormat): string {
  switch (format) {
    case 'csv':
      return 'text/csv;charset=utf-8';

    case 'markdown':
      return 'text/markdown;charset=utf-8';

    case 'json':
      return 'application/json;charset=utf-8';
  }
}

function downloadTextFile(
  filename: string,
  text: string,
  mimeType: string,
): void {
  const blob = new Blob([text], { type: mimeType });
  const url = URL.createObjectURL(blob);
  const anchor = document.createElement('a');

  anchor.href = url;
  anchor.download = filename;
  anchor.rel = 'noopener';

  document.body.appendChild(anchor);
  anchor.click();
  anchor.remove();

  window.setTimeout(() => URL.revokeObjectURL(url), 0);
}
```

```ts
export async function insertWriterQuickTemplate(
  id: WriterQuickTemplateId,
): Promise<InsertWriterQuickTemplateResult> {
  const template = getWriterQuickTemplate(id);
  const session = useSessionStore.getState();
  const writer = useWriterV2Store.getState();
  const sourceBytes = session.workingBytes ?? session.originalBytes;

  if (!sourceBytes || session.pageCount <= 0) {
    return {
      changed: false,
      message: 'Open a PDF before inserting a Writer template.',
    };
  }

  const { PDFDocument } = await import('pdf-lib');
  const pdf = await PDFDocument.load(sourceBytes);
  const pageIndex = Math.max(
    0,
    Math.min(session.viewState.currentPage - 1, pdf.getPageCount() - 1),
  );
  const page = pdf.getPage(pageIndex);
  const pageSize = page.getSize();

  const maxZIndex = writer.elements.reduce(
    (max, element) => Math.max(max, element.style.zIndex),
    0,
  );

  const element = createElementFromTemplate({
    template,
    documentFingerprint:
      writer.currentDocumentFingerprint ??
      session.documentKey ??
      session.fileName ??
      'writer-v2-unsaved-document',
    pageIndex,
    rect: centerRect(pageSize.width, pageSize.height, template.width, template.height),
    zIndex: maxZIndex + 1,
  });

  writer.dispatchWriterCommand({
    type: 'ADD_WRITER_ELEMENT',
    element,
  });

  return {
    changed: true,
    id: element.id,
    message: `Inserted Writer template "${template.label}".`,
  };
}

function createElementFromTemplate({
  template,
  documentFingerprint,
  pageIndex,
  rect,
  zIndex,
}: {
  template: WriterQuickTemplate;
  documentFingerprint: string;
  pageIndex: number;
  rect: PdfRect;
  zIndex: number;
}): WriterElementV2 {
  const input = {
    documentFingerprint,
    pageIndex,
    rect,
    text: template.text,
    zIndex,
  };

  const element = (() => {
    switch (template.elementType) {
      case 'textStamp':
        return createTextStampElement(input);

      case 'reviewLabel':
        return createReviewLabelElement(input);

      case 'richNote':
        return createRichNoteElement(input);

      case 'textCallout':
        return createTextCalloutElement({
          ...input,
          anchorPoint: {
            x: rect.x + rect.width + 24,
            y: rect.y + rect.height / 2,
          },
        });
    }
  })();

  const now = new Date().toISOString();

  return {
    ...element,
    plainText: template.text,
    style: {
      ...element.style,
      ...template.style,
      zIndex,
    },
    review: template.review
      ? {
          ...template.review,
          commentId: createReviewId(),
          createdAt: now,
          updatedAt: now,
        }
      : element.review,
  };
}

function centerRect(
  pageWidth: number,
  pageHeight: number,
  width: number,
  height: number,
): PdfRect {
  return {
    x: pageWidth / 2 - width / 2,
    y: pageHeight / 2 - height / 2,
    width,
    height,
  };
}

function createReviewId(): string {
  if (globalThis.crypto?.randomUUID) {
    return `review-${globalThis.crypto.randomUUID()}`;
  }

  return `review-${Date.now().toString(36)}-${Math.random()
    .toString(36)
    .slice(2)}`;
}
```

# 1. New file — search/filter engine

## `frontend/src/core/writer-v2/writerObjectSearch.ts`

```ts
import type {
  WriterElementV2,
  WriterReviewCategory,
  WriterReviewPriority,
  WriterReviewStatus,
} from './types';

export type WriterObjectSearchTypeFilter =
  | 'all'
  | WriterElementV2['type'];

export interface WriterObjectSearchQuery {
  text?: string;
  type?: WriterObjectSearchTypeFilter;
  page?: number | 'all';
  includeHidden?: boolean;
  reviewStatus?: WriterReviewStatus | 'all';
  reviewPriority?: WriterReviewPriority | 'all';
  reviewCategory?: WriterReviewCategory | 'all';
  onlyWithReview?: boolean;
}

export interface WriterObjectSearchResult {
  id: string;
  page: number;
  type: WriterElementV2['type'];
  label: string;
  text: string;
  hidden: boolean;
  locked: boolean;
  reviewStatus?: WriterReviewStatus;
  reviewPriority?: WriterReviewPriority;
  reviewCategory?: WriterReviewCategory;
}

export function searchWriterObjects(
  elements: WriterElementV2[],
  query: WriterObjectSearchQuery,
): WriterObjectSearchResult[] {
  const normalizedText = normalize(query.text ?? '');
  const type = query.type ?? 'all';
  const page = query.page ?? 'all';
  const includeHidden = query.includeHidden ?? false;
  const reviewStatus = query.reviewStatus ?? 'all';
  const reviewPriority = query.reviewPriority ?? 'all';
  const reviewCategory = query.reviewCategory ?? 'all';
  const onlyWithReview = query.onlyWithReview ?? false;

  return elements
    .filter((element) => includeHidden || !element.hidden)
    .filter((element) => type === 'all' || element.type === type)
    .filter((element) => page === 'all' || element.pageIndex + 1 === page)
    .filter((element) => !onlyWithReview || Boolean(element.review))
    .filter((element) => reviewStatus === 'all' || element.review?.status === reviewStatus)
    .filter((element) => reviewPriority === 'all' || element.review?.priority === reviewPriority)
    .filter((element) => reviewCategory === 'all' || element.review?.category === reviewCategory)
    .filter((element) => {
      if (!normalizedText) return true;

      const searchable = normalize([
        element.id,
        element.type,
        element.plainText ?? '',
        element.review?.status ?? '',
        element.review?.priority ?? '',
        element.review?.category ?? '',
        element.review?.author ?? '',
        element.review?.assignee ?? '',
        element.review?.dueDate ?? '',
      ].join(' '));

      return searchable.includes(normalizedText);
    })
    .map(toSearchResult)
    .sort((a, b) => {
      if (a.page !== b.page) return a.page - b.page;
      return a.label.localeCompare(b.label);
    });
}

export function writerObjectTypeLabel(type: WriterElementV2['type']): string {
  switch (type) {
    case 'richTextBox':
      return 'Text Box';

    case 'textCallout':
      return 'Callout';

    case 'textStamp':
      return 'Stamp';

    case 'reviewLabel':
      return 'Review Label';

    case 'richNote':
      return 'Note';

    case 'tableBox':
      return 'Table';

    case 'imageBox':
      return 'Image';

    default:
      return type;
  }
}

function toSearchResult(element: WriterElementV2): WriterObjectSearchResult {
  const text = normalizeDisplayText(element.plainText);

  return {
    id: element.id,
    page: element.pageIndex + 1,
    type: element.type,
    label: `${writerObjectTypeLabel(element.type)} - ${text || shortId(element.id)}`,
    text,
    hidden: element.hidden,
    locked: element.locked,
    reviewStatus: element.review?.status,
    reviewPriority: element.review?.priority,
    reviewCategory: element.review?.category,
  };
}

function normalize(value: string): string {
  return value.trim().toLowerCase();
}

function normalizeDisplayText(value: string | undefined): string {
  return (value ?? '')
    .replace(/\s+/g, ' ')
    .trim()
    .slice(0, 80);
}

function shortId(id: string): string {
  return id.length <= 8 ? id : id.slice(0, 8);
}
```

```tsx
        <label style={checkboxLabelStyle}>
          <input
            type="checkbox"
            checked={includeHidden}
            onChange={(event) => setIncludeHidden(event.currentTarget.checked)}
          />
          Include hidden objects
        </label>

        <button
          type="button"
          onClick={() => {
            setText('');
            setType('all');
            setPageFilter('all');
            setReviewStatus('all');
            setReviewPriority('all');
            setReviewCategory('all');
            setOnlyWithReview(false);
            setIncludeHidden(false);
          }}
          style={buttonStyle}
        >
          Clear filters
        </button>
      </div>

      <div
        role="list"
        aria-label="Writer object search results"
        style={{
          overflowY: 'auto',
          padding: 8,
          display: 'grid',
          gap: 6,
        }}
      >
        {results.length === 0 ? (
          <div
            style={{
              padding: 12,
              textAlign: 'center',
              color: '#94a3b8',
              fontSize: 12,
            }}
          >
            No Writer objects match the current filters.
          </div>
        ) : (
          results.map((result) => (
            <button
              key={result.id}
              type="button"
              role="listitem"
              onClick={() => jumpToObject(result.id, result.page)}
              aria-label={`Jump to ${result.label}`}
              style={{
                border: activeId === result.id
                  ? '1px solid #2563eb'
                  : '1px solid #e2e8f0',
                background: activeId === result.id
                  ? '#eff6ff'
                  : result.hidden
                    ? '#f8fafc'
                    : '#ffffff',
                borderRadius: 8,
                padding: 8,
                textAlign: 'left',
                cursor: 'pointer',
                opacity: result.hidden ? 0.65 : 1,
              }}
            >
              <div
                style={{
                  display: 'flex',
                  gap: 8,
                  justifyContent: 'space-between',
                  alignItems: 'center',
                  marginBottom: 3,
                }}
              >
                <span
                  style={{
                    fontSize: 11,
                    fontWeight: 800,
                    color: '#0f172a',
                    overflow: 'hidden',
                    textOverflow: 'ellipsis',
                    whiteSpace: 'nowrap',
                  }}
                >
                  {result.label}
                </span>
                <span
                  style={{
                    fontSize: 10,
                    color: result.page === currentPage ? '#2563eb' : '#64748b',
                    whiteSpace: 'nowrap',
                  }}
                >
                  p{result.page}
                </span>
              </div>

              <div
                style={{
                  fontSize: 10,
                  color: '#64748b',
                  display: 'flex',
                  gap: 6,
                  flexWrap: 'wrap',
                }}
              >
                <span>{writerObjectTypeLabel(result.type)}</span>
                {result.reviewStatus ? <span>Status: {result.reviewStatus}</span> : null}
                {result.reviewPriority ? <span>Priority: {result.reviewPriority}</span> : null}
                {result.reviewCategory ? <span>Category: {result.reviewCategory}</span> : null}
                {result.locked ? <span>Locked</span> : null}
                {result.hidden ? <span>Hidden</span> : null}
              </div>
            </button>
          ))
        )}
      </div>
    </aside>
  );
};
```
