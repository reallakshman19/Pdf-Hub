1. Patch registry execute() to never throw:

async execute(step, ctx, state) {
  const executor = this.executors.get(step.op);
  if (!executor) {
    return { status: 'error', message: `[MacroRegistry] No executor registered for op "${step.op}".`, sideEffects: [] };
  }
  try { return await executor(step, ctx, state); }
  catch (err) { return { status: 'error', message: err instanceof Error ? err.message : String(err), sideEffects: [] }; }
}
