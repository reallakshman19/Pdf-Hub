import { FileAdapter } from '@/adapters/file/FileAdapter';
import { PdfEditAdapter } from '@/adapters/pdf-edit/PdfEditAdapter';
import { bakeWriterElementsIntoPdf, type WriterExportWarning } from '@/core/writer/exporter';
import type { PdfAnnotation } from '@/core/annotations/types';
import type { PlacedElement } from '@/core/writer/types';

export interface ComposeCommittedPdfInput {
  baseBytes: Uint8Array;
  annotations: PdfAnnotation[];
  writerElements: PlacedElement[];
}

export interface ComposeCommittedPdfResult {
  bytes: Uint8Array;
  pageCount: number;
  documentKey: string;
  committedAnnotationCount: number;
  committedWriterElementCount: number;
  hadAnnotations: boolean;
  hadWriterElements: boolean;
  writerExportWarnings: WriterExportWarning[];
}

/**
 * Creates the exact PDF bytes that should be written to disk for Save/Export.
 *
 * Contract:
 * 1. Start from current working PDF bytes.
 * 2. Bake writer/page-builder elements first.
 * 3. Flatten annotations second so review markup remains visually above writer content.
 * 4. Recount pages from final bytes.
 * 5. Hash final bytes for the next durable documentKey.
 *
 * This function is intentionally pure from app-state perspective:
 * it does not mutate session store, annotation store, writer store, file handles,
 * dirty flags, or UI state.
 */
export async function composeCommittedPdf(
  input: ComposeCommittedPdfInput,
): Promise<ComposeCommittedPdfResult> {
  const annotations = Array.isArray(input.annotations) ? input.annotations : [];
  const writerElements = Array.isArray(input.writerElements) ? input.writerElements : [];

  let nextBytes = new Uint8Array(input.baseBytes);
  const writerExportWarnings: WriterExportWarning[] = [];

  if (writerElements.length > 0) {
    nextBytes = await bakeWriterElementsIntoPdf(nextBytes, writerElements, {
      preferSemantic: true,
      allowRasterFallback: true,
      collectWarnings: (warnings) => writerExportWarnings.push(...warnings),
    });
  }

  if (annotations.length > 0) {
    nextBytes = await PdfEditAdapter.exportWithAnnotations(nextBytes, annotations);
  }

  const pageCount = await PdfEditAdapter.countPages(nextBytes);
  const documentKey = await FileAdapter.hashBytes(nextBytes);

  return {
    bytes: new Uint8Array(nextBytes),
    pageCount,
    documentKey,
    committedAnnotationCount: annotations.length,
    committedWriterElementCount: writerElements.length,
    hadAnnotations: annotations.length > 0,
    hadWriterElements: writerElements.length > 0,
    writerExportWarnings,
  };
}

export async function makeCommittedDocumentKey(bytes: Uint8Array): Promise<string> {
  return FileAdapter.hashBytes(bytes);
}

function uniqSorted(values: number[]): number[] {
  return Array.from(new Set(values)).sort((a, b) => a - b);
}

function clampPage(pageNumber: number, pageCount: number): number {
  return Math.max(1, Math.min(pageNumber, Math.max(1, pageCount)));
}

function validSelectedPages(pages: number[], pageCount: number): number[] {
  return uniqSorted(pages.filter((page) => page >= 1 && page <= Math.max(1, pageCount)));
}

commitWorkingCopy: ({ bytes, pageCount, documentKey }) =>
  set((state) => ({
    documentKey: documentKey === undefined ? state.documentKey : documentKey,
    originalBytes: new Uint8Array(bytes),
    workingBytes: new Uint8Array(bytes),
    pageCount,
    isDirty: false,
    isDocumentDirty: false,
    isReviewDirty: false,
    isSessionDirty: false,
    selectedPages: validSelectedPages(state.selectedPages, pageCount),
    viewState: {
      ...state.viewState,
      currentPage: clampPage(state.viewState.currentPage, pageCount),
    },
    saveState: {
      status: 'saved',
      completedAt: Date.now(),
      message: 'Document saved',
    },
    annotationAutosaveState: {
      status: 'saved',
      pendingCount: 0,
      lastSavedAt: Date.now(),
    },
  })),

import { openDB } from 'idb';
import type { DBSchema, IDBPDatabase } from 'idb';
import type { PdfAnnotation } from './types';
import { error, debug } from '@/core/logger/service';

const DB_NAME = 'DocCraftAnnotations';
const STORE_NAME = 'annotations';
const DB_VERSION = 1;

interface AnnotationRecord {
  id: string;
  annotations: PdfAnnotation[];
  updatedAt: number;
}

interface AnnotationDbSchema extends DBSchema {
  annotations: {
    key: string;
    value: AnnotationRecord;
  };
}

export class AnnotationPersistenceError extends Error {
  readonly cause?: unknown;
  readonly operation: 'load' | 'save';

  constructor(
    operation: 'load' | 'save',
    message: string,
    cause?: unknown,
  ) {
    super(message);
    this.name = 'AnnotationPersistenceError';
    this.operation = operation;
    this.cause = cause;
  }
}

export const getDb = async (): Promise<IDBPDatabase<AnnotationDbSchema>> => {
  return openDB<AnnotationDbSchema>(DB_NAME, DB_VERSION, {
    upgrade(db) {
      if (!db.objectStoreNames.contains(STORE_NAME)) {
        db.createObjectStore(STORE_NAME, { keyPath: 'id' });
      }
    },
  });
};

export const saveAnnotations = async (
  documentKey: string,
  annotations: PdfAnnotation[],
): Promise<void> => {
  try {
    const db = await getDb();
    const tx = db.transaction(STORE_NAME, 'readwrite');

    await tx.store.put({
      id: documentKey,
      annotations,
      updatedAt: Date.now(),
    });

    await tx.done;

    debug('annotation', 'Annotation store flushed to IndexedDB', {
      count: annotations.length,
      documentKey,
    });
  } catch (err) {
    error('annotation', 'Failed to save annotations to IndexedDB', {
      error: String(err),
      documentKey,
      count: annotations.length,
    });

    throw new AnnotationPersistenceError(
      'save',
      'Failed to save annotations to local storage',
      err,
    );
  }
};

export const loadAnnotations = async (
  documentKey: string,
): Promise<PdfAnnotation[]> => {
  try {
    const db = await getDb();
    const result = await db.get(STORE_NAME, documentKey);

    const annotations = Array.isArray(result?.annotations)
      ? result.annotations
      : [];

    debug('annotation', 'Annotation store loaded from IndexedDB', {
      count: annotations.length,
      documentKey,
    });

    return annotations;
  } catch (err) {
    error('annotation', 'Failed to load annotations from IndexedDB', {
      error: String(err),
      documentKey,
    });

    throw new AnnotationPersistenceError(
      'load',
      'Failed to load annotations from local storage',
      err,
    );
  }
};

export function isAnnotationPersistenceError(
  err: unknown,
): err is AnnotationPersistenceError {
  return err instanceof AnnotationPersistenceError;
}

Core algorithm:
1. PDFDocument.load(baseBytes)
2. Embed Helvetica regular/bold/italic/boldItalic.
3. Sort elements by pageNumber then zIndex.
4. For image:
   - parse data URL/base64
   - embedPng/embedJpg
   - drawImage at top-to-PDF coordinates
5. For table:
   - parse JSON table data
   - compute column widths
   - draw rectangles with border/background
   - draw cell text with PDF text calls
6. For rich-text:
   - sanitize HTML
   - reject unsupported nodes for semantic export: table,img,svg,canvas,video,audio,iframe,object,embed,form,input,button,textarea,select
   - parse simple p/div/li/headings and b/strong/i/em/color/align
   - wrap text by font width
   - draw real PDF text
7. If semantic export fails and allowRasterFallback true:
   - render sanitized HTML/table to offscreen DOM
   - html2canvas
   - embed PNG fallback
   - collect warning
8. Return saved bytes.

Static anti-regression requirements:
- No container.innerHTML = element.content
- No ${cell.text} raw interpolation
- sanitizeWriterHtml must exist
- escapeHtml must exist
- drawTableElement must exist
- drawRichTextElement must exist

import { useAnnotationStore } from '@/core/annotations/store';
import { useWriterStore } from '@/core/writer/store';
import type { MacroMutableState, SideEffect } from './registry';
import type { MacroExecutionContext } from './types';

export interface MacroTransactionSnapshot {
  workingBytes: Uint8Array;
  pageCount: number;
  selectedPages: number[];
  outputFiles: { name: string; bytes: Uint8Array }[];
  annotations: ReturnType<typeof useAnnotationStore.getState>['annotations'];
  writerElements: ReturnType<typeof useWriterStore.getState>['elements'];
}

export function beginMacroTransaction(state: MacroMutableState): MacroTransactionSnapshot {
  return {
    workingBytes: new Uint8Array(state.workingBytes),
    pageCount: state.pageCount,
    selectedPages: [...state.selectedPages],
    outputFiles: state.outputFiles.map((file) => ({
      name: file.name,
      bytes: new Uint8Array(file.bytes),
    })),
    annotations: useAnnotationStore.getState().annotations.map((annotation) => ({
      ...annotation,
      rect: { ...annotation.rect },
      data: { ...annotation.data },
    })),
    writerElements: useWriterStore.getState().elements.map((element) => ({
      ...element,
      styles: { ...element.styles },
    })),
  };
}

export function rollbackMacroTransaction(
  state: MacroMutableState,
  snapshot: MacroTransactionSnapshot,
): void {
  state.workingBytes = new Uint8Array(snapshot.workingBytes);
  state.pageCount = snapshot.pageCount;
  state.selectedPages = [...snapshot.selectedPages];
  state.outputFiles = snapshot.outputFiles.map((file) => ({
    name: file.name,
    bytes: new Uint8Array(file.bytes),
  }));

  useAnnotationStore.setState({
    annotations: snapshot.annotations,
    selectedAnnotationIds: [],
    activeAnnotationId: null,
  });

  useWriterStore.setState({
    elements: snapshot.writerElements,
    selectedIds: [],
    activeId: null,
    activeTool: 'select',
  });
}

export function applyMacroSideEffects(
  sideEffects: SideEffect[],
  state: MacroMutableState,
): void {
  for (const effect of sideEffects) {
    if (effect.type === 'bytes_updated') state.workingBytes = effect.bytes;

    if (effect.type === 'page_count_changed') {
      state.pageCount = effect.newCount;
      state.selectedPages = state.selectedPages.filter(
        (page) => page >= 1 && page <= effect.newCount,
      );
    }

    if (effect.type === 'output_file') {
      state.outputFiles.push({
        name: effect.name,
        bytes: effect.bytes,
      });
    }
  }
}

export function buildMacroRollbackMessage(
  ctx: MacroExecutionContext,
  stepIndex: number,
  op: string,
  message?: string,
): string {
  return [
    `Macro rolled back for ${ctx.fileName}`,
    `at step ${stepIndex + 1}`,
    `(${op})`,
    message ? `: ${message}` : '',
  ].join(' ');
}

export const SaveStatusIndicator: React.FC = () => {
  const {
    isDirty,
    isDocumentDirty,
    isReviewDirty,
    isSessionDirty,
    saveState,
    annotationAutosaveState,
    fileName,

    isDocumentDirty,
    isReviewDirty,
    isSessionDirty,
    saveStatus: saveState.status,
    annotationAutosaveStatus: annotationAutosaveState.status,
    writerElementCount,
    annotationCount,
  });

  const Icon = status.icon;

  return (
    <Tooltip content={status.tooltip}>
      <div
        className={[
          'flex items-center gap-1.5 rounded-md px-2 py-1 text-xs',
          status.className,
        ].join(' ')}
        aria-label={status.tooltip}
      >
        <Icon className={status.spin ? 'h-3.5 w-3.5 animate-spin' : 'h-3.5 w-3.5'} />
        <span className="hidden md:inline">{status.label}</span>
        {fileName && (
          <span className="hidden xl:inline max-w-[180px] truncate opacity-70">
            {fileName}
          </span>
        )}
      </div>
    </Tooltip>
  );
};

function resolveStatus(input: {
  isDirty: boolean;
  isDocumentDirty: boolean;
  isReviewDirty: boolean;
  isSessionDirty: boolean;
  saveStatus: string;
  annotationAutosaveStatus: string;
  writerElementCount: number;
  annotationCount: number;
}) {
  if (input.saveStatus === 'saving') {
    return {
      label: 'Saving…',
      tooltip: 'Saving document bytes, annotations, and writer elements.',
      icon: Loader2,
      spin: true,
      className: 'bg-blue-50 text-blue-700 dark:bg-blue-950/40 dark:text-blue-300',
    };
  }

  if (input.saveStatus === 'exporting') {
    return {
      label: 'Exporting…',
      tooltip: 'Building an exported PDF. The current session will remain editable.',
      icon: Loader2,
      spin: true,
      className: 'bg-blue-50 text-blue-700 dark:bg-blue-950/40 dark:text-blue-300',
    };
  }

  if (input.saveStatus === 'failed') {
    return {
      label: 'Save failed',
      tooltip: 'The last save/export failed. Your in-session changes are still present.',
      icon: AlertCircle,
      className: 'bg-red-50 text-red-700 dark:bg-red-950/40 dark:text-red-300',
    };
  }

  if (input.annotationAutosaveStatus === 'failed') {
    return {
      label: 'Autosave failed',
      tooltip:
        'Local annotation autosave failed. Save/export the PDF or retry after checking browser storage permissions.',
      icon: WifiOff,
      className: 'bg-red-50 text-red-700 dark:bg-red-950/40 dark:text-red-300',
    };
  }

  if (input.annotationAutosaveStatus === 'saving') {
    return {
      label: 'Autosaving…',
      tooltip: 'Saving annotation overlay state to local browser storage.',
      icon: Loader2,
      spin: true,
      className: 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300',
    };
  }

  if (input.isDirty) {
    const details: string[] = [];
    if (input.isDocumentDirty) details.push('document bytes changed');
    if (input.isReviewDirty || input.annotationCount > 0) {
      details.push(`${input.annotationCount} annotation${input.annotationCount === 1 ? '' : 's'}`);
    }
    if (input.isSessionDirty || input.writerElementCount > 0) {
      details.push(`${input.writerElementCount} writer element${input.writerElementCount === 1 ? '' : 's'}`);
    }

    return {
      label: 'Unsaved',
      tooltip: details.length > 0 ? `Unsaved changes: ${details.join(', ')}.` : 'There are unsaved changes.',
      icon: Clock,
      className: 'bg-amber-50 text-amber-800 dark:bg-amber-950/40 dark:text-amber-300',
    };
  }

  if (input.saveStatus === 'saved' || input.annotationAutosaveStatus === 'saved') {
    return {
      label: 'Saved',
      tooltip: 'Current document state is saved.',
      icon: CheckCircle2,
      className: 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300',
    };
  }

  return {
    label: 'Ready',
    tooltip: 'No active save operation.',
    icon: Save,
    className: 'bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-300',
  };
}

import { FileAdapter } from '@/adapters/file/FileAdapter';
import { composeCommittedPdf } from '@/core/document/commitPipeline';
import type { PdfAnnotation } from '@/core/annotations/types';
import type { PlacedElement } from '@/core/writer/types';

export async function emergencyDownloadCommittedPdf(input: {
  workingBytes: Uint8Array;
  fileName: string;
  annotations: PdfAnnotation[];
  writerElements: PlacedElement[];
}): Promise<void> {
  const committed = await composeCommittedPdf({
    baseBytes: input.workingBytes,
    annotations: input.annotations,
    writerElements: input.writerElements,
  });

  const safeName = input.fileName.replace(/\.pdf$/i, '') + '-recovery.pdf';
  FileAdapter.downloadBytes(committed.bytes, safeName);
}
