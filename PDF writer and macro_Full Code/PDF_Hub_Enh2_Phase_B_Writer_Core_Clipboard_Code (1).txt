import type {
  PlacedElement,
  WriterContentModel,
  WriterImageModel,
  WriterRichTextBlock,
  WriterRichTextModel,
  WriterTableCell,
  WriterTableColumn,
  WriterTableModel,
  WriterTableRow,
  WriterTextRun,
} from './types';

let idCounter = 0;

export function createWriterId(prefix = 'writer'): string {
  idCounter += 1;
  return `${prefix}-${Date.now().toString(36)}-${idCounter.toString(36)}`;
}

export function createTextRun(text: string, patch: Partial<WriterTextRun> = {}): WriterTextRun {
  return {
    id: createWriterId('run'),
    text,
    ...patch,
  };
}

export function createRichTextBlock(
  text = '',
  patch: Partial<WriterRichTextBlock> = {},
): WriterRichTextBlock {
  return {
    id: createWriterId('block'),
    type: 'paragraph',
    runs: [createTextRun(text)],
    ...patch,
  };
}

export function createEmptyRichTextModel(text = ''): WriterRichTextModel {
  return {
    kind: 'rich-text',
    version: 1,
    blocks: [createRichTextBlock(text)],
  };
}

export function createRichTextModelFromPlainText(text: string): WriterRichTextModel {
  const lines = String(text ?? '').replace(/\r\n/g, '\n').split('\n');

  return {
    kind: 'rich-text',
    version: 1,
    blocks: lines.length > 0
      ? lines.map((line) => createRichTextBlock(line))
      : [createRichTextBlock('')],
  };
}

export function createDefaultTableModel(rows = 3, columns = 3): WriterTableModel {
  const cols: WriterTableColumn[] = Array.from({ length: columns }, (_, index) => ({
    id: createWriterId('col'),
    label: columnLabel(index),
  }));

  const tableRows: WriterTableRow[] = Array.from({ length: rows }, (_, rowIndex) => ({
    id: createWriterId('row'),
    cells: cols.map((col, colIndex) => ({
      id: createWriterId('cell'),
      text: rowIndex === 0 ? col.label : '',
      style: rowIndex === 0 ? { bold: true } : undefined,
    })),
  }));

  return {
    kind: 'table',
    version: 1,
    columns: cols,
    rows: tableRows,
    style: {
      borderColor: '#d1d5db',
      borderWidth: 1,
      headerBackgroundColor: '#f1f5f9',
      cellPadding: 6,
      fontSize: 11,
      fontFamily: 'Arial',
    },
  };
}

export function createTableModelFromMatrix(matrix: string[][]): WriterTableModel {
  const safeMatrix = matrix.length > 0 ? matrix : [['']];
  const columnCount = Math.max(...safeMatrix.map((row) => row.length), 1);

  const columns: WriterTableColumn[] = Array.from({ length: columnCount }, (_, index) => ({
    id: createWriterId('col'),
    label: columnLabel(index),
  }));

  const rows: WriterTableRow[] = safeMatrix.map((row, rowIndex) => ({
    id: createWriterId('row'),
    cells: columns.map((_, colIndex) => ({
      id: createWriterId('cell'),
      text: row[colIndex] ?? '',
      style: rowIndex === 0 ? { bold: true } : undefined,
    })),
  }));

  return {
    kind: 'table',
    version: 1,
    columns,
    rows,
    style: {
      borderColor: '#d1d5db',
      borderWidth: 1,
      headerBackgroundColor: '#f1f5f9',
      cellPadding: 6,
      fontSize: 11,
      fontFamily: 'Arial',
    },
  };
}

export function createImageModel(input: {

  return {

    naturalHeight: input.naturalHeight,
    alt: input.alt,

    borderWidth: 0,

export function normalizeWriterContentModel(model: WriterContentModel): WriterContentModel {
  if (model.kind === 'rich-text') return normalizeRichTextModel(model);
  if (model.kind === 'table') return normalizeTableModel(model);
  if (model.kind === 'image') return normalizeImageModel(model);
  return model;
}

export function normalizeRichTextModel(model: WriterRichTextModel): WriterRichTextModel {
  return {
    kind: 'rich-text',
    version: 1,
    blocks: Array.isArray(model.blocks) && model.blocks.length > 0
      ? model.blocks.map((block) => ({
          id: block.id || createWriterId('block'),
          type: block.type || 'paragraph',
          align: block.align,
          level: block.level,
          runs: Array.isArray(block.runs) && block.runs.length > 0
            ? block.runs.map((run) => ({
                id: run.id || createWriterId('run'),
                text: String(run.text ?? ''),
                bold: run.bold,
                italic: run.italic,
                underline: run.underline,
                strike: run.strike,
                color: run.color,
                backgroundColor: run.backgroundColor,
                fontSize: run.fontSize,
                fontFamily: run.fontFamily,
                superscript: run.superscript,
                subscript: run.subscript,
              }))
            : [createTextRun('')],
        }))
      : [createRichTextBlock('')],
  };
}

export function normalizeTableModel(model: WriterTableModel): WriterTableModel {
  const columns = Array.isArray(model.columns) && model.columns.length > 0
    ? model.columns.map((col, index) => ({
        id: col.id || createWriterId('col'),
        label: col.label || columnLabel(index),
        width: col.width,
      }))
    : [{ id: createWriterId('col'), label: 'A' }];

  const rows = Array.isArray(model.rows) && model.rows.length > 0
    ? model.rows.map((row) => ({
        id: row.id || createWriterId('row'),
        cells: columns.map((_, index) => normalizeTableCell(row.cells?.[index])),
      }))
    : [
        {
          id: createWriterId('row'),
          cells: columns.map(() => normalizeTableCell(undefined)),
        },
      ];

  return {
    kind: 'table',
    version: 1,
    columns,
    rows,
    style: model.style,
  };
}

export function normalizeTableCell(cell: WriterTableCell | undefined): WriterTableCell {
  return {
    id: cell?.id || createWriterId('cell'),
    text: String(cell?.text ?? ''),
    richText: cell?.richText ? normalizeRichTextModel(cell.richText) : undefined,
    colspan: cell?.colspan,
    rowspan: cell?.rowspan,
    style: cell?.style,
  };
}

export function normalizeImageModel(model: WriterImageModel): WriterImageModel {
  return {
    kind: 'image',
    version: 1,
    src: String(model.src ?? ''),
    mimeType: model.mimeType || inferMimeTypeFromDataUrl(model.src) || 'image/png',
    naturalWidth: model.naturalWidth,
    naturalHeight: model.naturalHeight,
    alt: model.alt,
    crop: model.crop,
    rotation: Number.isFinite(model.rotation) ? model.rotation : 0,
    opacity: Number.isFinite(model.opacity) ? model.opacity : 1,
    borderColor: model.borderColor,
    borderWidth: Number.isFinite(model.borderWidth) ? model.borderWidth : 0,
  };
}

export function normalizeWriterElement(element: PlacedElement): PlacedElement {
  const contentModel = element.contentModel
    ? normalizeWriterContentModel(element.contentModel)
    : undefined;

  return {
    ...element,
    id: element.id || createWriterId('element'),
    pageNumber: Math.max(1, Math.floor(element.pageNumber || 1)),
    x: finiteOr(element.x, 0),
    y: finiteOr(element.y, 0),
    width: Math.max(1, finiteOr(element.width, 160)),
    height: Math.max(1, finiteOr(element.height, 60)),
    content: String(element.content ?? ''),
    contentModel,
    zIndex: finiteOr(element.zIndex, 0),
    hidden: Boolean(element.hidden),
    locked: Boolean(element.locked),
  };
}

function finiteOr(value: unknown, fallback: number): number {
  return typeof value === 'number' && Number.isFinite(value) ? value : fallback;
}

function inferMimeTypeFromDataUrl(src: string | undefined): string | undefined {
  if (!src) return undefined;
  const match = /^data:([^;,]+)[;,]/i.exec(src);
  return match?.[1];
}

function columnLabel(index: number): string {
  let n = index + 1;
  let out = '';

  while (n > 0) {
    const rem = (n - 1) % 26;
    out = String.fromCharCode(65 + rem) + out;
    n = Math.floor((n - 1) / 26);
  }

  return out;
}

import type {
  PlacedElement,
  WriterContentModel,
  WriterImageModel,
  WriterRichTextModel,
  WriterTableModel,
} from './types';
import {
  createEmptyRichTextModel,
  normalizeWriterContentModel,
} from './content-model';

export function writerContentModelToLegacyContent(model: WriterContentModel): string {
  if (model.kind === 'rich-text') return richTextToPlainText(model);
  if (model.kind === 'table') return tableModelToLegacyJson(model);
  if (model.kind === 'image') return model.src;
  return '';
}

export function syncElementContentFromModel(element: PlacedElement): PlacedElement {
  if (!element.contentModel) return element;

  return {
    ...element,
    contentModel: normalizeWriterContentModel(element.contentModel),
    content: writerContentModelToLegacyContent(element.contentModel),
  };
}

export function tryParseLegacyContentModel(content: string): WriterContentModel | null {
  if (!content) return null;

  try {
    const parsed = JSON.parse(content) as unknown;

    if (
      parsed &&
      typeof parsed === 'object' &&
      'kind' in parsed &&
      'version' in parsed
    ) {
      return normalizeWriterContentModel(parsed as WriterContentModel);
    }

    if (
      parsed &&
      typeof parsed === 'object' &&
      Array.isArray((parsed as { rows?: unknown }).rows) &&
      Array.isArray((parsed as { columns?: unknown }).columns)
    ) {
      return normalizeWriterContentModel({
        kind: 'table',
        version: 1,
        ...(parsed as WriterTableModel),
      });
    }
  } catch {
    return null;
  }

  return null;
}

export function legacyContentToRichTextModel(content: string): WriterRichTextModel {
  return createEmptyRichTextModel(String(content ?? ''));
}

export function richTextToPlainText(model: WriterRichTextModel): string {
  return model.blocks
    .map((block) => block.runs.map((run) => run.text).join(''))
    .join('\n');
}

export function tableModelToLegacyJson(model: WriterTableModel): string {
  return JSON.stringify({
    columns: model.columns,
    rows: model.rows,
    style: model.style,
  });
}

export function imageModelToLegacyContent(model: WriterImageModel): string {
  return model.src;
}

export function safeJsonStringify(value: unknown): string {
  try {
    return JSON.stringify(value);
  } catch {
    return '';
  }
}

import type { PlacedElement } from './types';
import {
  createEmptyRichTextModel,
  normalizeWriterElement,
} from './content-model';
import {
  syncElementContentFromModel,
  tryParseLegacyContentModel,
} from './content-serialization';

export function migratePlacedElementToContentModel(element: PlacedElement): PlacedElement {
  if (element.contentModel) {
    return syncElementContentFromModel(normalizeWriterElement(element));
  }

  const parsed = tryParseLegacyContentModel(element.content);

  if (parsed) {
    return syncElementContentFromModel(
      normalizeWriterElement({
        ...element,
        contentModel: parsed,
        sourceMeta: element.sourceMeta ?? {
          source: 'legacy',
          importedAt: new Date().toISOString(),
        },
      }),
    );
  }

  if (element.type === 'rich-text' || element.type === 'text') {
    return syncElementContentFromModel(
      normalizeWriterElement({
        ...element,
        type: 'rich-text',
        contentModel: createEmptyRichTextModel(element.content ?? ''),
        sourceMeta: element.sourceMeta ?? {
          source: 'legacy',
          importedAt: new Date().toISOString(),
        },
      }),
    );
  }

  return normalizeWriterElement(element);
}

export function migratePlacedElementsToContentModel(elements: PlacedElement[]): PlacedElement[] {
  return elements.map(migratePlacedElementToContentModel);
}

export function ensureWriterElementCompatibility(element: PlacedElement): PlacedElement {
  return migratePlacedElementToContentModel(element);
}

const BLOCKED_TAGS = new Set([
  'script',
  'iframe',
  'object',
  'embed',
  'link',
  'meta',
  'base',
  'form',
  'input',
  'button',
  'textarea',
  'select',
  'style',
]);

const SAFE_STYLE_PROPS = new Set([
  'font-weight',
  'font-style',
  'text-decoration',
  'text-align',
  'color',
  'background-color',
  'font-size',
  'font-family',
  'line-height',
]);

export function sanitizeClipboardHtml(html: string): string {
  if (!html) return '';

  if (typeof document === 'undefined') {
    return stripDangerousHtmlFallback(html);
  }

  const template = document.createElement('template');
  template.innerHTML = html;

  const nodes = Array.from(template.content.querySelectorAll('*'));

  for (const node of nodes) {
    const tagName = node.tagName.toLowerCase();

    if (BLOCKED_TAGS.has(tagName)) {
      node.remove();
      continue;
    }

    for (const attr of Array.from(node.attributes)) {
      const name = attr.name.toLowerCase();
      const value = attr.value.trim();

      if (name.startsWith('on')) {
        node.removeAttribute(attr.name);
        continue;
      }

      if ((name === 'href' || name === 'src') && !isSafeClipboardUrl(value)) {
        node.removeAttribute(attr.name);
        continue;
      }

      if (name === 'style') {
        const safeStyle = sanitizeStyleAttribute(value);

        if (safeStyle) {
          node.setAttribute('style', safeStyle);
        } else {
          node.removeAttribute('style');
        }
      }
    }
  }

  return template.innerHTML;
}

export function escapeClipboardHtml(value: unknown): string {
  return String(value ?? '')
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#039;');
}

export function isSafeClipboardUrl(value: string): boolean {
  if (/^javascript:/i.test(value)) return false;
  if (/^data:text\/html/i.test(value)) return false;

  return (
    /^https?:\/\//i.test(value) ||
    /^data:image\/(png|jpeg|jpg|webp);base64,/i.test(value) ||
    value.startsWith('#')
  );
}

export function sanitizeStyleAttribute(style: string): string {
  return style
    .split(';')
    .map((rule) => rule.trim())
    .filter(Boolean)
    .map((rule) => {
      const [rawProp, ...rawValue] = rule.split(':');
      const prop = rawProp?.trim().toLowerCase();
      const value = rawValue.join(':').trim();

      if (!prop || !SAFE_STYLE_PROPS.has(prop)) return '';
      if (/url\s*\(/i.test(value)) return '';
      if (/expression\s*\(/i.test(value)) return '';
      if (/javascript:/i.test(value)) return '';

      return `${prop}: ${value.replace(/[<>"'`]/g, '')}`;
    })
    .filter(Boolean)
    .join('; ');
}

function stripDangerousHtmlFallback(html: string): string {
  return html
    .replace(/<script[\s\S]*?<\/script>/gi, '')
    .replace(/\son[a-z]+\s*=\s*(['"]).*?\1/gi, '')
    .replace(/\s(href|src)\s*=\s*(['"])\s*javascript:[\s\S]*?\2/gi, '');
}

import {
  createTableModelFromMatrix,
} from '../content-model';
import type { WriterTableModel } from '../types';
import { sanitizeClipboardHtml } from './sanitizeClipboardHtml';

export function parseClipboardTableFromText(text: string): WriterTableModel | null {
  const normalized = String(text ?? '').replace(/\r\n/g, '\n').trimEnd();

  if (!normalized) return null;

  const delimiter = normalized.includes('\t') ? '\t' : normalized.includes(',') ? ',' : null;

  if (!delimiter) return null;

  const rows = normalized
    .split('\n')
    .map((line) => parseDelimitedLine(line, delimiter));

  if (rows.length < 1) return null;

  const hasMultipleColumns = rows.some((row) => row.length > 1);
  if (!hasMultipleColumns) return null;

  return createTableModelFromMatrix(rows);
}

export function parseClipboardTableFromHtml(html: string): WriterTableModel | null {
  const sanitized = sanitizeClipboardHtml(html);

  if (typeof document === 'undefined') return null;

  const template = document.createElement('template');
  template.innerHTML = sanitized;

  const table = template.content.querySelector('table');
  if (!table) return null;

  const rows = Array.from(table.querySelectorAll('tr')).map((tr) => {
    return Array.from(tr.querySelectorAll('th,td')).map((cell) =>
      normalizeCellText(cell.textContent ?? ''),
    );
  });

  const nonEmptyRows = rows.filter((row) => row.length > 0);
  if (nonEmptyRows.length === 0) return null;

  return createTableModelFromMatrix(nonEmptyRows);
}

function parseDelimitedLine(line: string, delimiter: string): string[] {
  if (delimiter === '\t') return line.split('\t').map(normalizeCellText);
  return parseCsvLine(line).map(normalizeCellText);
}

function parseCsvLine(line: string): string[] {
  const out: string[] = [];
  let current = '';
  let inQuotes = false;

  for (let i = 0; i < line.length; i += 1) {
    const ch = line[i];

    if (ch === '"') {
      if (inQuotes && line[i + 1] === '"') {
        current += '"';
        i += 1;
      } else {
        inQuotes = !inQuotes;
      }
      continue;
    }

    if (ch === ',' && !inQuotes) {
      out.push(current);
      current = '';
      continue;
    }

    current += ch;
  }

  out.push(current);
  return out;
}

function normalizeCellText(value: string): string {
  return value.replace(/\s+/g, ' ').trim();
}

import type {
  WriterRichTextBlock,
  WriterRichTextModel,
  WriterTextRun,
} from '../types';
import {
  createRichTextBlock,
  createTextRun,
} from '../content-model';
import { sanitizeClipboardHtml } from './sanitizeClipboardHtml';

export function parseClipboardRichTextFromHtml(html: string): WriterRichTextModel {
  const sanitized = sanitizeClipboardHtml(html);

  if (typeof document === 'undefined') {
    return {
      kind: 'rich-text',
      version: 1,
      blocks: [createRichTextBlock(stripTags(sanitized))],
    };
  }

  const template = document.createElement('template');
  template.innerHTML = sanitized;

  const blocks: WriterRichTextBlock[] = [];

  const blockNodes = Array.from(
    template.content.querySelectorAll('p,div,h1,h2,h3,li,blockquote'),
  );

  if (blockNodes.length === 0) {
    const text = template.content.textContent ?? '';
    return {
      kind: 'rich-text',
      version: 1,
      blocks: [createRichTextBlock(text.trim())],
    };
  }

  for (const node of blockNodes) {
    const block = nodeToBlock(node);
    if (block.runs.some((run) => run.text.length > 0)) {
      blocks.push(block);
    }
  }

  return {
    kind: 'rich-text',
    version: 1,
    blocks: blocks.length > 0 ? blocks : [createRichTextBlock('')],
  };
}

function nodeToBlock(node: Element): WriterRichTextBlock {
  const tag = node.tagName.toLowerCase();
  const runs = collectRuns(node);

  return {
    id: createRichTextBlock('').id,
    type: tag === 'li'
      ? 'bullet'
      : tag === 'blockquote'
        ? 'quote'
        : tag === 'h1' || tag === 'h2' || tag === 'h3'
          ? 'heading'
          : 'paragraph',
    level: tag === 'h1' ? 1 : tag === 'h2' ? 2 : tag === 'h3' ? 3 : undefined,
    runs: runs.length > 0 ? runs : [createTextRun(node.textContent ?? '')],
    align: readAlign(node),
  };
}

function collectRuns(root: Element): WriterTextRun[] {
  const runs: WriterTextRun[] = [];

  function walk(node: Node, inherited: Partial<WriterTextRun>) {
    if (node.nodeType === Node.TEXT_NODE) {
      const text = node.textContent ?? '';
      if (text) runs.push(createTextRun(text, inherited));
      return;
    }

    if (node.nodeType !== Node.ELEMENT_NODE) return;

    const el = node as Element;
    const tag = el.tagName.toLowerCase();
    const style = (el as HTMLElement).style;

    const next: Partial<WriterTextRun> = {
      ...inherited,
      bold: inherited.bold || tag === 'strong' || tag === 'b' || style.fontWeight === 'bold',
      italic: inherited.italic || tag === 'em' || tag === 'i' || style.fontStyle === 'italic',
      underline: inherited.underline || tag === 'u' || style.textDecoration.includes('underline'),
      strike: inherited.strike || tag === 's' || style.textDecoration.includes('line-through'),
      color: style.color || inherited.color,
      backgroundColor: style.backgroundColor || inherited.backgroundColor,
      fontFamily: style.fontFamily || inherited.fontFamily,
      fontSize: style.fontSize ? parseFloat(style.fontSize) : inherited.fontSize,
    };

    for (const child of Array.from(el.childNodes)) {
      walk(child, next);
    }
  }

  walk(root, {});
  return runs;
}

function readAlign(node: Element): WriterRichTextBlock['align'] {
  const align = (node as HTMLElement).style.textAlign;
  if (align === 'center' || align === 'right' || align === 'justify') return align;
  return 'left';
}

function stripTags(value: string): string {
  return value.replace(/<[^>]+>/g, '').trim();
}

import type { WriterImageModel } from '../types';
import { createImageModel } from '../content-model';

export async function blobToClipboardImage(blob: Blob, alt?: string): Promise<WriterImageModel> {
  const src = await blobToDataUrl(blob);
  const size = await readImageSize(src);

  return createImageModel({
    src,
    mimeType: blob.type || inferMimeType(src) || 'image/png',
    naturalWidth: size.width,
    naturalHeight: size.height,
    alt,
  });
}

export function isClipboardImageType(type: string): boolean {
  return /^image\/(png|jpeg|jpg|webp|gif)$/i.test(type);
}

function blobToDataUrl(blob: Blob): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();

    reader.onload = () => resolve(String(reader.result ?? ''));
    reader.onerror = () => reject(reader.error ?? new Error('Failed to read image blob.'));
    reader.readAsDataURL(blob);
  });
}

function readImageSize(src: string): Promise<{ width: number; height: number }> {
  if (typeof Image === 'undefined') {
    return Promise.resolve({ width: 0, height: 0 });
  }

  return new Promise((resolve) => {
    const img = new Image();

    img.onload = () => resolve({ width: img.naturalWidth, height: img.naturalHeight });
    img.onerror = () => resolve({ width: 0, height: 0 });
    img.src = src;
  });
}

function inferMimeType(src: string): string | undefined {
  const match = /^data:([^;,]+)[;,]/i.exec(src);
  return match?.[1];
}

import type { WriterClipboardPayload } from './types';
import { parseClipboardTableFromHtml, parseClipboardTableFromText } from './parseClipboardTable';
import { parseClipboardRichTextFromHtml } from './parseClipboardHtml';
import { blobToClipboardImage, isClipboardImageType } from './parseClipboardImage';

export async function readNavigatorClipboardPayload(): Promise<WriterClipboardPayload> {
  if (typeof navigator === 'undefined' || !navigator.clipboard) {
    return {
      kind: 'unsupported',
      warning: 'Navigator clipboard API is not available.',
    };
  }

  const navClipboard = navigator.clipboard as Clipboard & {
    read?: () => Promise<ClipboardItem[]>;
  };

  if (typeof navClipboard.read === 'function') {
    try {
      const items = await navClipboard.read();

      for (const item of items) {
        const imageType = item.types.find(isClipboardImageType);

        if (imageType) {
          const blob = await item.getType(imageType);
          return {
            kind: 'image',
            image: await blobToClipboardImage(blob),
            mimeType: imageType,
          };
        }
      }

      for (const item of items) {
        if (item.types.includes('text/html')) {
          const blob = await item.getType('text/html');
          const html = await blob.text();
          const table = parseClipboardTableFromHtml(html);

          if (table) {
            return {
              kind: 'table',
              html,
              table,
              mimeType: 'text/html',
            };
          }

          return {
            kind: 'html',
            html,
            richText: parseClipboardRichTextFromHtml(html),
            mimeType: 'text/html',
          };
        }
      }

      for (const item of items) {
        if (item.types.includes('text/plain')) {
          const blob = await item.getType('text/plain');
          return textPayload(await blob.text());
        }
      }
    } catch (err) {
      return {
        kind: 'unsupported',
        warning: err instanceof Error ? err.message : String(err),
      };
    }
  }

  if (typeof navigator.clipboard.readText === 'function') {
    try {
      return textPayload(await navigator.clipboard.readText());
    } catch (err) {
      return {
        kind: 'unsupported',
        warning: err instanceof Error ? err.message : String(err),
      };
    }
  }

  return {
    kind: 'unsupported',
    warning: 'Clipboard read is not supported.',
  };
}

export function payloadFromClipboardEvent(event: ClipboardEvent): WriterClipboardPayload {
  const data = event.clipboardData;

  if (!data) {
    return { kind: 'empty' };
  }

  for (const item of Array.from(data.items)) {
    if (item.kind === 'file' && isClipboardImageType(item.type)) {
      const file = item.getAsFile();

      if (file) {
        return {
          kind: 'unsupported',
          warning: 'Image clipboard item detected. Use async image paste handler.',
          mimeType: item.type,
        };
      }
    }
  }

  const html = data.getData('text/html');

  if (html) {
    const table = parseClipboardTableFromHtml(html);

    if (table) {
      return {
        kind: 'table',
        html,
        table,
        mimeType: 'text/html',
      };
    }

    return {
      kind: 'html',
      html,
      richText: parseClipboardRichTextFromHtml(html),
      mimeType: 'text/html',
    };
  }

  const text = data.getData('text/plain');
  return textPayload(text);
}

function textPayload(text: string): WriterClipboardPayload {
  if (!text) {
    return { kind: 'empty' };
  }

  const table = parseClipboardTableFromText(text);

  if (table) {
    return {
      kind: 'table',
      plainText: text,
      table,
      mimeType: 'text/plain',
    };
  }

  return {
    kind: 'plain-text',
    plainText: text,
    mimeType: 'text/plain',
  };
}

import type { PlacedElement } from '../types';
import {
  createEmptyRichTextModel,
  createWriterId,
} from '../content-model';
import {
  syncElementContentFromModel,
} from '../content-serialization';
import type {
  ClipboardElementPlacement,
  ClipboardToElementsResult,
  WriterClipboardPayload,
} from './types';

export function clipboardPayloadToElements(
  payload: WriterClipboardPayload,
  placement: ClipboardElementPlacement,
): ClipboardToElementsResult {
  const base = {
    id: createWriterId('element'),
    pageNumber: placement.pageNumber,
    x: placement.x,
    y: placement.y,
    zIndex: placement.zIndex ?? 0,
    locked: false,
    hidden: false,
  };

  const importedAt = new Date().toISOString();

  if (payload.kind === 'table' && payload.table) {
    const element: PlacedElement = syncElementContentFromModel({
      ...base,
      type: 'table',
      width: placement.width ?? placement.defaultWidth ?? 420,
      height: placement.height ?? placement.defaultHeight ?? 160,
      content: '',
      contentModel: payload.table,
      sourceMeta: {
        source: 'clipboard-table',
        mimeType: payload.mimeType,
        importedAt,
        warning: payload.warning,
      },
      style: {
        opacity: 1,
      },
    });

    return { elements: [element], warning: payload.warning };
  }

  if ((payload.kind === 'html' || payload.kind === 'plain-text') && (payload.richText || payload.plainText)) {
    const richText = payload.richText ?? createEmptyRichTextModel(payload.plainText ?? '');

    const element: PlacedElement = syncElementContentFromModel({
      ...base,
      type: 'rich-text',
      width: placement.width ?? placement.defaultWidth ?? 320,
      height: placement.height ?? placement.defaultHeight ?? 96,
      content: '',
      contentModel: richText,
      sourceMeta: {
        source: payload.kind === 'html' ? 'clipboard-html' : 'clipboard-text',
        mimeType: payload.mimeType,
        importedAt,
        warning: payload.warning,
      },
      style: {
        fontSize: 12,
        fontFamily: 'Arial',
        color: '#111827',
        backgroundColor: 'transparent',
        opacity: 1,
      },
    });

    return { elements: [element], warning: payload.warning };
  }

  if (payload.kind === 'image' && payload.image) {
    const naturalWidth = payload.image.naturalWidth || 240;
    const naturalHeight = payload.image.naturalHeight || 160;
    const targetWidth = placement.width ?? Math.min(naturalWidth, placement.defaultWidth ?? 320);
    const ratio = naturalHeight > 0 && naturalWidth > 0 ? naturalHeight / naturalWidth : 0.66;
    const targetHeight = placement.height ?? Math.max(1, Math.round(targetWidth * ratio));

    const element: PlacedElement = syncElementContentFromModel({
      ...base,
      type: 'image',
      width: targetWidth,
      height: targetHeight,
      content: '',
      contentModel: payload.image,
      sourceMeta: {
        source: 'clipboard-image',
        mimeType: payload.mimeType,
        importedAt,
        warning: payload.warning,
      },
      style: {
        opacity: payload.image.opacity ?? 1,
      },
    });

    return { elements: [element], warning: payload.warning };
  }

  return {
    elements: [],
    warning: payload.warning ?? 'Clipboard payload did not produce writer elements.',
  };
}

import type { PlacedElement } from '../types';
import {
  clipboardPayloadToElements,
} from './clipboardToElements';
import {
  payloadFromClipboardEvent,
  readNavigatorClipboardPayload,
} from './readClipboard';
import type {
  ClipboardElementPlacement,
  ClipboardToElementsResult,
} from './types';

export interface WriterPasteControllerOptions {
  getPlacement: () => ClipboardElementPlacement;
  addElements: (elements: PlacedElement[]) => void;
  onWarning?: (message: string) => void;
}

export function createWriterPasteController(options: WriterPasteControllerOptions) {
  return {
    async pasteFromNavigator(): Promise<ClipboardToElementsResult> {
      const payload = await readNavigatorClipboardPayload();
      const result = clipboardPayloadToElements(payload, options.getPlacement());

      if (result.warning) options.onWarning?.(result.warning);
      if (result.elements.length > 0) options.addElements(result.elements);

      return result;
    },

    pasteFromClipboardEvent(event: ClipboardEvent): ClipboardToElementsResult {
      if (shouldIgnorePasteTarget(event.target)) {
        return {
          elements: [],
          warning: 'Paste ignored because target is editable.',
        };
      }

      const payload = payloadFromClipboardEvent(event);
      const result = clipboardPayloadToElements(payload, options.getPlacement());

      if (result.elements.length > 0) {
        event.preventDefault();
        options.addElements(result.elements);
      }

      if (result.warning) options.onWarning?.(result.warning);

      return result;
    },
  };
}

export function shouldIgnorePasteTarget(target: EventTarget | null): boolean {
  if (!(target instanceof HTMLElement)) return false;

  const tag = target.tagName.toLowerCase();

  return (
    target.isContentEditable ||
    tag === 'input' ||
    tag === 'textarea' ||
    tag === 'select' ||
    Boolean(target.closest('[data-writer-allow-native-paste="true"]')) ||
    Boolean(target.closest('[data-macro-builder="true"]'))
  );
}
