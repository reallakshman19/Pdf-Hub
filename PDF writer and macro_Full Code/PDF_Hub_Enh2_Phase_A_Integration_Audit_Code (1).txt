```js
function main() {
  section('PDF Hub Enh2 Integration Audit');

  report.push(`Working directory: ${root}`);
  report.push(`Generated: ${new Date().toISOString()}`);
  report.push('');

  auditPackageJson();
  auditExpectedRoots();

  auditDuplicateSuspiciousFiles();

function auditPackageJson() {
  section('Package scripts');

  const packagePath = path.join(root, 'package.json');

  if (!fs.existsSync(packagePath)) {
    fail('package.json not found in frontend root.');
    return;
  }

  const pkg = JSON.parse(fs.readFileSync(packagePath, 'utf8'));
  const scripts = pkg.scripts ?? {};

  const scriptChecks = [
    'typecheck',
    'test',
    'build',
    'audit:writer',
    'audit:macro',
    'certify:writer',
    'certify:macro',
    'audit:enh2',
    'audit:enh2:shared',
    'errors:enh2',
    'certify:enh2',
  ];

  for (const script of scriptChecks) {
    if (scripts[script]) pass(`script ${script}: ${scripts[script]}`);
    else warn(`missing script ${script}`);
  }
}

function auditExpectedRoots() {
  section('Expected source roots');

  for (const rel of EXPECTED_SOURCE_ROOTS) {
    if (exists(rel)) {
      const fileCount = countFiles(rel);
      pass(`${rel} exists (${fileCount} file(s))`);
    } else {
      warn(`${rel} missing`);
    }
  }
}

function auditRequiredGroups() {
  section('Enhancement module groups');

  for (const group of REQUIRED_GROUPS) {
    const present = group.requiredAny.filter(exists);
    const missing = group.requiredAny.filter((rel) => !exists(rel));

    if (present.length > 0) {
      pass(`${group.name}: ${present.length}/${group.requiredAny.length} expected file(s) present`);
      for (const rel of present) report.push(`  PRESENT ${rel}`);
    } else {
      fail(`${group.name}: no expected files found`);
    }

    for (const rel of missing) report.push(`  MISSING ${rel}`);
    report.push('');
  }
}

function auditDuplicateSuspiciousFiles() {
  section('Duplicate/suspicious files');

  const all = listFiles('src');
  const suspicious = all.filter((file) => /(\.tmp$|\.bak$| copy\.| copy |old|backup|duplicate)/i.test(file));

  if (suspicious.length === 0) {
    pass('No suspicious duplicate/temp source files found.');
    return;
  }

  warn(`${suspicious.length} suspicious duplicate/temp file(s) found.`);
  for (const file of suspicious) report.push(`  ${file}`);
}

function auditExpectedDocs() {
  section('Expected integration documents');

  for (const rel of EXPECTED_DOCS) {
    if (exists(rel)) {
      pass(`${rel} exists`);
    } else {
      warn(`${rel} missing; script will create placeholder.`);
      const title = rel.replace(/_/g, ' ').replace(/\.md$/i, '');
      writePlaceholderDoc(rel, title);
    }
  }
}

function writeReports() {
  section('Summary');

  report.push(`Warnings: ${warnings.length}`);
  report.push(`Failures: ${failures.length}`);
  report.push('');

  if (warnings.length > 0) {
    report.push('Warnings:');
    for (const warning of warnings) report.push(`- ${warning}`);
    report.push('');
  }

  if (failures.length > 0) {
    report.push('Failures:');
    for (const failure of failures) report.push(`- ${failure}`);
    report.push('');
  }

  const auditPath = path.join(root, 'ENH2_INTEGRATION_AUDIT.md');
  fs.writeFileSync(auditPath, report.join('\n') + '\n', 'utf8');

  const errorRegisterPath = path.join(root, 'ENH2_ERROR_REGISTER.md');
  if (!fs.existsSync(errorRegisterPath)) {
    fs.writeFileSync(
      errorRegisterPath,
      [
        '# ENH2 Error Register',
        '',
        '| ID | Module | Type | File | Error | Severity | Status |',
        '|---|---|---|---|---|---|---|',
        '| ENH2-ERR-001 | Pending | Baseline | TBD | Populate after typecheck/test/build | TBD | Open |',
        '',
      ].join('\n'),
      'utf8',
    );
  }

  console.log(`Wrote ${auditPath}`);
}

function section(title) {
  report.push('');
  report.push(`## ${title}`);
  report.push('');
}

function exists(rel) {
  return fs.existsSync(path.join(root, rel));
}

function listFiles(relRoot) {
  const absRoot = path.join(root, relRoot);
  if (!fs.existsSync(absRoot)) return [];

  const out = [];
  walk(absRoot, out);
  return out.map((abs) => path.relative(root, abs).replaceAll(path.sep, '/'));
}

function walk(dir, out) {
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    const abs = path.join(dir, entry.name);
    if (entry.isDirectory()) walk(abs, out);
    else out.push(abs);
  }
}

function countFiles(relRoot) {
  return listFiles(relRoot).length;
}

function pass(message) {
  report.push(`PASS: ${message}`);
}

function warn(message) {
  warnings.push(message);
  report.push(`WARN: ${message}`);
}

function fail(message) {
  failures.push(message);
  report.push(`FAIL: ${message}`);
}

function writePlaceholderDoc(rel, title) {
  const abs = path.join(root, rel);
  if (fs.existsSync(abs)) return;

  fs.writeFileSync(
    abs,
    [
      `# ${title}`,
      '',
      'Status: Placeholder generated by enh2-integration-audit.mjs.',
      '',
      'This file must be completed during the corresponding implementation phase.',
      '',
    ].join('\n'),
    'utf8',
  );
}
```

```js
import fs from 'node:fs';
import path from 'node:path';
import process from 'node:process';

const root = process.cwd();

const CANDIDATE_PATTERNS = [
  {
    id: 'PDF_BYTE_UPDATE',
    description: 'Potential PDF byte update functions/state',
    regex: /(set.*Pdf.*Bytes|update.*Pdf.*Bytes|finalBytes|workingBytes|pdfBytes|documentBytes)/i,
  },
  {
    id: 'DIRTY_STATE',
    description: 'Potential dirty-state functions/state',
    regex: /(mark.*Dirty|set.*Dirty|dirty|isDirty|unsaved)/i,
  },
  {
    id: 'SAVE_EXPORT',
    description: 'Potential save/export functions',
    regex: /(savePdf|save.*Document|exportPdf|export.*Document|downloadPdf|flatten)/i,
  },
  {
    id: 'PAGE_SELECTION',
    description: 'Potential selected page/current page state',
    regex: /(selectedPages|setSelectedPages|currentPage|setCurrentPage|pageCount)/i,
  },
  {
    id: 'WRITER_OVERLAY',
    description: 'Potential writer overlay integration',
    regex: /(WriterOverlay|WriterRibbon|WriterSidePanel|PlacedElement|contentModel)/i,
  },
  {
    id: 'MACRO_PANEL',
    description: 'Potential macro panel/dropdown integration',
    regex: /(MacroPanel|MacroDropdown|executeMacroRecipe|buildMacroExecutionContext)/i,
  },
];

const report = [];
main();

function main() {
  report.push('# ENH2 Shared Contracts Audit');
  report.push('');
  report.push(`Generated: ${new Date().toISOString()}`);
  report.push(`Working directory: ${root}`);
  report.push('');

  const files = listFiles('src').filter(isSourceFile);

  for (const pattern of CANDIDATE_PATTERNS) {
    report.push(`## ${pattern.id} — ${pattern.description}`);
    report.push('');
    let count = 0;

    for (const file of files) {
      const abs = path.join(root, file);
      const text = safeRead(abs);
      if (!text) continue;
      const lines = text.split(/\r?\n/);

      lines.forEach((line, index) => {
        if (pattern.regex.test(line)) {
          count += 1;
          report.push(`- ${file}:${index + 1}: ${line.trim().slice(0, 220)}`);
        }
      });
    }

    if (count === 0) report.push('- No candidates found.');
    report.push('');
  }

  report.push('## Required Final Contract Decisions');
  report.push('');
  report.push('| Contract | Chosen implementation | File/function | Status |');
  report.push('|---|---|---|---|');
  report.push('| PDF byte source | TBD | TBD | Open |');
  report.push('| PDF byte update function | TBD | TBD | Open |');
  report.push('| Dirty-state marker | TBD | TBD | Open |');
  report.push('| Page count source | TBD | TBD | Open |');
  report.push('| Selected pages source | TBD | TBD | Open |');
  report.push('| Writer save/commit path | TBD | TBD | Open |');
  report.push('| Macro apply result path | TBD | TBD | Open |');
  report.push('| Render cache refresh | TBD | TBD | Open |');
  report.push('');

  fs.writeFileSync(path.join(root, 'ENH2_SHARED_CONTRACTS.md'), report.join('\n') + '\n', 'utf8');
  console.log('Wrote ENH2_SHARED_CONTRACTS.md');
}

function listFiles(relRoot) {
  const absRoot = path.join(root, relRoot);
  if (!fs.existsSync(absRoot)) return [];
  const out = [];
  walk(absRoot, out);
  return out.map((abs) => path.relative(root, abs).replaceAll(path.sep, '/'));
}

function walk(dir, out) {
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    const abs = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      if (entry.name === 'node_modules' || entry.name === 'dist' || entry.name === 'build') continue;
      walk(abs, out);
    } else {
      out.push(abs);
    }
  }
}

function isSourceFile(file) {
  return /\.(ts|tsx|js|jsx|mjs|cjs)$/.test(file);
}

function safeRead(abs) {
  try {
    return fs.readFileSync(abs, 'utf8');
  } catch {
    return '';
  }
}
```

```js
import fs from 'node:fs';

const input = fs.readFileSync(0, 'utf8');
const lines = input.split(/\r?\n/);
const rows = [];
let id = 1;

for (const line of lines) {
  if (/error TS\d+:/i.test(line) || /failed/i.test(line) || /error/i.test(line)) {
    rows.push({ id: `ENH2-ERR-${String(id).padStart(3, '0')}`, line });
    id += 1;
  }
}

const md = [
  '# ENH2 Error Register',
  '',
  '| ID | Module | Type | File | Error | Severity | Status |',
  '|---|---|---|---|---|---|---|',
  ...rows.map((row) => {
    const fileMatch = row.line.match(/([A-Za-z0-9_\-./\\]+\.(ts|tsx|js|jsx|mjs|cjs))/);
    const file = fileMatch ? fileMatch[1].replaceAll('\\', '/') : 'TBD';
    const module = file.includes('/writer/') || file.includes('components/writer') ? 'Writer'
      : file.includes('/macro/') || file.includes('components/macro') ? 'Macro'
      : 'Shared/App';
    const type = /TS\d+/.test(row.line) ? 'TypeScript'
      : /test/i.test(row.line) ? 'Test'
      : /build/i.test(row.line) ? 'Build'
      : 'General';
    return `| ${row.id} | ${module} | ${type} | ${file} | ${escapeCell(row.line.trim())} | TBD | Open |`;
  }),
  '',
];

fs.writeFileSync('ENH2_ERROR_REGISTER.md', md.join('\n'), 'utf8');
console.log(`Wrote ENH2_ERROR_REGISTER.md with ${rows.length} row(s).`);

function escapeCell(value) {
  return value.replaceAll('|', '\\|').slice(0, 500);
}
```
