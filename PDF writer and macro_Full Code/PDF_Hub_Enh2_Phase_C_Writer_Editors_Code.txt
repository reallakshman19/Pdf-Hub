import type {
  WriterRichTextBlock,
  WriterRichTextModel,
  WriterTextRun,
} from './types';

export function richTextModelToHtml(model: WriterRichTextModel): string {
  return model.blocks.map(blockToHtml).join('');
}

export function richTextModelToPlainText(model: WriterRichTextModel): string {
  return model.blocks
    .map((block) => block.runs.map((run) => run.text).join(''))
    .join('\n');
}

export function htmlToRichTextModel(html: string): WriterRichTextModel {
  if (typeof document === 'undefined') {
    return plainTextToRichTextModel(stripTags(html));
  }

  const template = document.createElement('template');
  template.innerHTML = sanitizeWriterHtml(html);

  const blockNodes = Array.from(
    template.content.querySelectorAll('p,div,h1,h2,h3,li,blockquote'),
  );

  if (blockNodes.length === 0) {
    return plainTextToRichTextModel(template.content.textContent ?? '');
  }

  return {
    kind: 'rich-text',
    version: 1,
    blocks: blockNodes.map((node, index) => elementToBlock(node, index)),
  };
}

export function plainTextToRichTextModel(text: string): WriterRichTextModel {
  const lines = String(text ?? '').replace(/\r\n/g, '\n').split('\n');

  return {
    kind: 'rich-text',
    version: 1,
    blocks: lines.map((line, index) => ({
      id: `block-${index + 1}`,
      type: 'paragraph',
      runs: [
        {
          id: `run-${index + 1}`,
          text: line,
        },
      ],
      align: 'left',
    })),
  };
}

export function sanitizeWriterHtml(html: string): string {
  if (!html) return '';

  if (typeof document === 'undefined') {
    return stripDangerousHtmlFallback(html);
  }

  const template = document.createElement('template');
  template.innerHTML = html;

  const blockedTags = new Set([
    'script',
    'iframe',
    'object',
    'embed',
    'link',
    'meta',
    'base',
    'form',
    'input',
    'button',
    'textarea',
    'select',
    'style',
  ]);

  const safeStyleProps = new Set([
    'font-weight',
    'font-style',
    'text-decoration',
    'text-align',
    'color',
    'background-color',
    'font-size',
    'font-family',
    'line-height',
  ]);

  for (const node of Array.from(template.content.querySelectorAll('*'))) {
    const tag = node.tagName.toLowerCase();

    if (blockedTags.has(tag)) {
      node.remove();
      continue;
    }

    for (const attr of Array.from(node.attributes)) {
      const name = attr.name.toLowerCase();
      const value = attr.value.trim();

      if (name.startsWith('on')) {
        node.removeAttribute(attr.name);
        continue;
      }

      if ((name === 'href' || name === 'src') && isUnsafeUrl(value)) {
        node.removeAttribute(attr.name);
        continue;
      }

      if (name === 'style') {
        const safeStyle = value
          .split(';')
          .map((rule) => rule.trim())
          .filter(Boolean)
          .map((rule) => {
            const [rawProp, ...rawValue] = rule.split(':');
            const prop = rawProp?.trim().toLowerCase();
            const styleValue = rawValue.join(':').trim();

            if (!prop || !safeStyleProps.has(prop)) return '';
            if (/url\s*\(/i.test(styleValue)) return '';
            if (/expression\s*\(/i.test(styleValue)) return '';
            if (/javascript:/i.test(styleValue)) return '';

            return `${prop}: ${styleValue.replace(/[<>"'`]/g, '')}`;
          })
          .filter(Boolean)
          .join('; ');

        if (safeStyle) node.setAttribute('style', safeStyle);
        else node.removeAttribute('style');
      }
    }
  }

  return template.innerHTML;
}

function blockToHtml(block: WriterRichTextBlock): string {
  const body = block.runs.map(runToHtml).join('');
  const style = block.align ? ` style="text-align: ${escapeAttr(block.align)}"` : '';

  if (block.type === 'heading') {
    const level = block.level ?? 2;
    const tag = level === 1 ? 'h1' : level === 3 ? 'h3' : 'h2';
    return `<${tag}${style}>${body}</${tag}>`;
  }

  if (block.type === 'bullet') return `<ul><li${style}>${body}</li></ul>`;
  if (block.type === 'numbered') return `<ol><li${style}>${body}</li></ol>`;
  if (block.type === 'quote') return `<blockquote${style}>${body}</blockquote>`;

  return `<p${style}>${body}</p>`;
}

function runToHtml(run: WriterTextRun): string {
  const styles: string[] = [];

  if (run.color) styles.push(`color: ${escapeStyleValue(run.color)}`);
  if (run.backgroundColor) styles.push(`background-color: ${escapeStyleValue(run.backgroundColor)}`);
  if (run.fontSize) styles.push(`font-size: ${Number(run.fontSize)}px`);
  if (run.fontFamily) styles.push(`font-family: ${escapeStyleValue(run.fontFamily)}`);

  let html = escapeHtml(run.text);

  if (run.bold) html = `<strong>${html}</strong>`;
  if (run.italic) html = `<em>${html}</em>`;
  if (run.underline) html = `<u>${html}</u>`;
  if (run.strike) html = `<s>${html}</s>`;
  if (run.superscript) html = `<sup>${html}</sup>`;
  if (run.subscript) html = `<sub>${html}</sub>`;

  if (styles.length > 0) {
    html = `<span style="${styles.join('; ')}">${html}</span>`;
  }

  return html;
}

function elementToBlock(node: Element, index: number): WriterRichTextBlock {
  const tag = node.tagName.toLowerCase();
  const runs = collectRuns(node);

  return {
    id: `block-${index + 1}`,
    type: tag === 'li'
      ? 'bullet'
      : tag === 'blockquote'
        ? 'quote'
        : tag === 'h1' || tag === 'h2' || tag === 'h3'
          ? 'heading'
          : 'paragraph',
    level: tag === 'h1' ? 1 : tag === 'h2' ? 2 : tag === 'h3' ? 3 : undefined,
    align: readAlign(node),
    runs: runs.length > 0 ? runs : [{ id: `run-${index + 1}`, text: node.textContent ?? '' }],
  };
}

function collectRuns(root: Element): WriterTextRun[] {
  const runs: WriterTextRun[] = [];

  function walk(node: Node, inherited: Partial<WriterTextRun>) {
    if (node.nodeType === Node.TEXT_NODE) {
      const text = node.textContent ?? '';
      if (text) {
        runs.push({
          id: `run-${runs.length + 1}`,
          text,
          ...inherited,
        });
      }
      return;
    }

    if (node.nodeType !== Node.ELEMENT_NODE) return;

    const el = node as HTMLElement;
    const tag = el.tagName.toLowerCase();

    const style = el.style;

    const next: Partial<WriterTextRun> = {
      ...inherited,
      bold:
        inherited.bold ||
        tag === 'b' ||
        tag === 'strong' ||
        style.fontWeight === 'bold' ||
        Number(style.fontWeight) >= 600,
      italic: inherited.italic || tag === 'i' || tag === 'em' || style.fontStyle === 'italic',
      underline: inherited.underline || tag === 'u' || style.textDecoration.includes('underline'),
      strike: inherited.strike || tag === 's' || style.textDecoration.includes('line-through'),
      color: style.color || inherited.color,
      backgroundColor: style.backgroundColor || inherited.backgroundColor,
      fontFamily: style.fontFamily || inherited.fontFamily,
      fontSize: style.fontSize ? Number.parseFloat(style.fontSize) : inherited.fontSize,
      superscript: inherited.superscript || tag === 'sup',
      subscript: inherited.subscript || tag === 'sub',
    };

    for (const child of Array.from(el.childNodes)) {
      walk(child, next);
    }
  }

  walk(root, {});
  return runs;
}

function readAlign(node: Element): WriterRichTextBlock['align'] {
  const align = (node as HTMLElement).style.textAlign;

  if (align === 'center' || align === 'right' || align === 'justify') return align;
  return 'left';
}

function isUnsafeUrl(value: string): boolean {
  return /^javascript:/i.test(value) || /^data:text\/html/i.test(value);
}

function stripTags(value: string): string {
  return value.replace(/<[^>]+>/g, '').trim();
}

function stripDangerousHtmlFallback(html: string): string {
  return html
    .replace(/<script[\s\S]*?<\/script>/gi, '')
    .replace(/\son[a-z]+\s*=\s*(['"]).*?\1/gi, '')
    .replace(/\s(href|src)\s*=\s*(['"])\s*javascript:[\s\S]*?\2/gi, '');
}

function escapeHtml(value: unknown): string {
  return String(value ?? '')
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#039;');
}

function escapeAttr(value: unknown): string {
  return escapeHtml(value).replaceAll('`', '');
}

function escapeStyleValue(value: unknown): string {
  return String(value ?? '').replace(/[<>"'`]/g, '');
}

import type {
  WriterRichTextBlock,
  WriterRichTextBlockType,
  WriterRichTextModel,
  WriterTextRun,
} from './types';
import {
  createTextRun,
} from './content-model';

export interface RichTextSelectionRef {
  blockIndex: number;
  runIndex?: number;
}

export function updateRichTextPlainText(
  model: WriterRichTextModel,
  text: string,
): WriterRichTextModel {
  const lines = String(text ?? '').replace(/\r\n/g, '\n').split('\n');

  return {
    ...model,
    blocks: lines.map((line, index) => {
      const existing = model.blocks[index];

      return {
        id: existing?.id ?? `block-${index + 1}`,
        type: existing?.type ?? 'paragraph',
        align: existing?.align ?? 'left',
        level: existing?.level,
        runs: [
          {
            ...(existing?.runs[0] ?? createTextRun('')),
            text: line,
          },
        ],
      };
    }),
  };
}

export function setBlockType(
  model: WriterRichTextModel,
  blockIndex: number,
  type: WriterRichTextBlockType,
): WriterRichTextModel {
  return updateBlock(model, blockIndex, (block) => ({
    ...block,
    type,
    level: type === 'heading' ? block.level ?? 2 : undefined,
  }));
}

export function setBlockAlign(
  model: WriterRichTextModel,
  blockIndex: number,
  align: WriterRichTextBlock['align'],
): WriterRichTextModel {
  return updateBlock(model, blockIndex, (block) => ({
    ...block,
    align,
  }));
}

export function setHeadingLevel(
  model: WriterRichTextModel,
  blockIndex: number,
  level: 1 | 2 | 3,
): WriterRichTextModel {
  return updateBlock(model, blockIndex, (block) => ({
    ...block,
    type: 'heading',
    level,
  }));
}

export function toggleRunMark(
  model: WriterRichTextModel,
  ref: RichTextSelectionRef,
  mark: 'bold' | 'italic' | 'underline' | 'strike',
): WriterRichTextModel {
  return updateRun(model, ref, (run) => ({
    ...run,
    [mark]: !run[mark],
  }));
}

export function setRunColor(
  model: WriterRichTextModel,
  ref: RichTextSelectionRef,
  color: string,
): WriterRichTextModel {
  return updateRun(model, ref, (run) => ({
    ...run,
    color,
  }));
}

export function setRunFontSize(
  model: WriterRichTextModel,
  ref: RichTextSelectionRef,
  fontSize: number,
): WriterRichTextModel {
  return updateRun(model, ref, (run) => ({
    ...run,
    fontSize: Math.max(1, fontSize),
  }));
}

export function setRunFontFamily(
  model: WriterRichTextModel,
  ref: RichTextSelectionRef,
  fontFamily: string,
): WriterRichTextModel {
  return updateRun(model, ref, (run) => ({
    ...run,
    fontFamily,
  }));
}

export function insertBlockAfter(
  model: WriterRichTextModel,
  blockIndex: number,
  text = '',
): WriterRichTextModel {
  const blocks = [...model.blocks];

  blocks.splice(blockIndex + 1, 0, {
    id: `block-${Date.now().toString(36)}`,
    type: 'paragraph',
    align: 'left',
    runs: [createTextRun(text)],
  });

  return {
    ...model,
    blocks,
  };
}

export function removeBlock(
  model: WriterRichTextModel,
  blockIndex: number,
): WriterRichTextModel {
  const blocks = model.blocks.filter((_, index) => index !== blockIndex);

  return {
    ...model,
    blocks: blocks.length > 0
      ? blocks
      : [
          {
            id: `block-${Date.now().toString(36)}`,
            type: 'paragraph',
            align: 'left',
            runs: [createTextRun('')],
          },
        ],
  };
}

export function mergeAdjacentRuns(model: WriterRichTextModel): WriterRichTextModel {
  return {
    ...model,
    blocks: model.blocks.map((block) => ({
      ...block,
      runs: mergeRuns(block.runs),
    })),
  };
}

function updateBlock(
  model: WriterRichTextModel,
  blockIndex: number,
  updater: (block: WriterRichTextBlock) => WriterRichTextBlock,
): WriterRichTextModel {
  return {
    ...model,
    blocks: model.blocks.map((block, index) =>
      index === blockIndex ? updater(block) : block,
    ),
  };
}

function updateRun(
  model: WriterRichTextModel,
  ref: RichTextSelectionRef,
  updater: (run: WriterTextRun) => WriterTextRun,
): WriterRichTextModel {
  return updateBlock(model, ref.blockIndex, (block) => ({
    ...block,
    runs: block.runs.map((run, index) =>
      index === (ref.runIndex ?? 0) ? updater(run) : run,
    ),
  }));
}

function mergeRuns(runs: WriterTextRun[]): WriterTextRun[] {
  const merged: WriterTextRun[] = [];

  for (const run of runs) {
    const previous = merged[merged.length - 1];

    if (previous && sameRunStyle(previous, run)) {
      previous.text += run.text;
    } else {
      merged.push({ ...run });
    }
  }

  return merged;
}

function sameRunStyle(a: WriterTextRun, b: WriterTextRun): boolean {
  return (
    a.bold === b.bold &&
    a.italic === b.italic &&
    a.underline === b.underline &&
    a.strike === b.strike &&
    a.color === b.color &&
    a.backgroundColor === b.backgroundColor &&
    a.fontSize === b.fontSize &&
    a.fontFamily === b.fontFamily &&
    a.superscript === b.superscript &&
    a.subscript === b.subscript
  );
}

export interface WriterTableSelection {
  startRow: number;
  startCol: number;
  endRow: number;
  endCol: number;
}

export interface WriterTableCellRef {
  row: number;
  col: number;
}

export function normalizeTableSelection(selection: WriterTableSelection): WriterTableSelection {
  return {
    startRow: Math.min(selection.startRow, selection.endRow),
    endRow: Math.max(selection.startRow, selection.endRow),
    startCol: Math.min(selection.startCol, selection.endCol),
    endCol: Math.max(selection.startCol, selection.endCol),
  };
}

export function isCellInSelection(
  cell: WriterTableCellRef,
  selection: WriterTableSelection | null,
): boolean {
  if (!selection) return false;

  const normalized = normalizeTableSelection(selection);

  return (
    cell.row >= normalized.startRow &&
    cell.row <= normalized.endRow &&
    cell.col >= normalized.startCol &&
    cell.col <= normalized.endCol
  );
}

export function createSingleCellSelection(row: number, col: number): WriterTableSelection {
  return {
    startRow: row,
    endRow: row,
    startCol: col,
    endCol: col,
  };
}

export function expandSelectionToCell(
  selection: WriterTableSelection,
  row: number,
  col: number,
): WriterTableSelection {
  return {
    ...selection,
    endRow: row,
    endCol: col,
  };
}

export function selectionToRefs(selection: WriterTableSelection): WriterTableCellRef[] {
  const normalized = normalizeTableSelection(selection);
  const refs: WriterTableCellRef[] = [];

  for (let row = normalized.startRow; row <= normalized.endRow; row += 1) {

import type {
  WriterTableCell,
  WriterTableColumn,
  WriterTableModel,
  WriterTableRow,
} from './types';

let idCounter = 0;

function id(prefix: string): string {
  idCounter += 1;
  return `${prefix}-${Date.now().toString(36)}-${idCounter.toString(36)}`;
}

export function updateTableCellText(
  table: WriterTableModel,
  rowIndex: number,
  colIndex: number,
  text: string,
): WriterTableModel {
  return {
    ...table,
    rows: table.rows.map((row, r) => ({
      ...row,
      cells: row.cells.map((cell, c) =>
        r === rowIndex && c === colIndex
          ? { ...cell, text }
          : cell,
      ),
    })),
  };
}

export function insertTableRow(
  table: WriterTableModel,
  atIndex: number,
): WriterTableModel {
  const safeIndex = Math.max(0, Math.min(atIndex, table.rows.length));
  const row = createEmptyRow(table.columns.length);

  return {
    ...table,
    rows: [
      ...table.rows.slice(0, safeIndex),
      row,
      ...table.rows.slice(safeIndex),
    ],
  };
}

export function deleteTableRow(
  table: WriterTableModel,
  rowIndex: number,
): WriterTableModel {
  if (table.rows.length <= 1) return table;

  return {
    ...table,
    rows: table.rows.filter((_, index) => index !== rowIndex),
  };
}

export function insertTableColumn(
  table: WriterTableModel,
  atIndex: number,
): WriterTableModel {
  const safeIndex = Math.max(0, Math.min(atIndex, table.columns.length));
  const column: WriterTableColumn = {
    id: id('col'),
    label: nextColumnLabel(table.columns.length),
  };

  return {
    ...table,
    columns: [
      ...table.columns.slice(0, safeIndex),
      column,
      ...table.columns.slice(safeIndex),
    ],
    rows: table.rows.map((row) => ({
      ...row,
      cells: [
        ...row.cells.slice(0, safeIndex),
        createEmptyCell(),
        ...row.cells.slice(safeIndex),
      ],
    })),
  };
}

export function deleteTableColumn(
  table: WriterTableModel,
  colIndex: number,
): WriterTableModel {
  if (table.columns.length <= 1) return table;

  return {
    ...table,
    columns: table.columns.filter((_, index) => index !== colIndex),
    rows: table.rows.map((row) => ({
      ...row,
      cells: row.cells.filter((_, index) => index !== colIndex),
    })),
  };
}

export function pasteMatrixIntoTable(
  table: WriterTableModel,
  startRow: number,
  startCol: number,
  matrix: string[][],
): WriterTableModel {
  let next = table;

  const requiredRows = startRow + matrix.length;
  while (next.rows.length < requiredRows) {
    next = insertTableRow(next, next.rows.length);
  }

  const requiredCols = startCol + Math.max(0, ...matrix.map((row) => row.length));
  while (next.columns.length < requiredCols) {
    next = insertTableColumn(next, next.columns.length);
  }

  const rows = next.rows.map((row) => ({
    ...row,
    cells: row.cells.map((cell) => ({ ...cell })),
  }));

  matrix.forEach((matrixRow, r) => {
    matrixRow.forEach((value, c) => {
      rows[startRow + r].cells[startCol + c].text = value;
    });
  });

  return {
    ...next,
    rows,
  };
}

export function tableToMatrix(table: WriterTableModel): string[][] {
  return table.rows.map((row) => row.cells.map((cell) => cell.text));
}

export function matrixToTsv(matrix: string[][]): string {
  return matrix.map((row) => row.join('\t')).join('\n');
}

function createEmptyRow(columnCount: number): WriterTableRow {
  return {
    id: id('row'),
    cells: Array.from({ length: columnCount }, () => createEmptyCell()),
  };
}

function createEmptyCell(): WriterTableCell {
  return {
    id: id('cell'),
    text: '',
  };
}

function nextColumnLabel(index: number): string {
  let n = index + 1;
  let out = '';

  while (n > 0) {
    const rem = (n - 1) % 26;
    out = String.fromCharCode(65 + rem) + out;
    n = Math.floor((n - 1) / 26);
  }

  return out;
}

import type {
  WriterImageModel,
} from './types';
import {
  createImageModel,
} from './content-model';

export async function fileToWriterImageModel(file: File): Promise<WriterImageModel> {
  if (!isSupportedWriterImageMime(file.type)) {
    throw new Error(`Unsupported image type: ${file.type || 'unknown'}`);
  }

  const src = await blobToDataUrl(file);
  const size = await readImageSize(src);

  return createImageModel({
    src,
    mimeType: file.type,
    naturalWidth: size.width,
    naturalHeight: size.height,
    alt: file.name,
  });
}

export function isSupportedWriterImageMime(mimeType: string): boolean {
  return /^image\/(png|jpeg|jpg|webp)$/i.test(mimeType);
}

export function extractImageFilesFromDataTransfer(dataTransfer: DataTransfer): File[] {
  return Array.from(dataTransfer.files).filter((file) =>
    isSupportedWriterImageMime(file.type),
  );
}

function blobToDataUrl(blob: Blob): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();

    reader.onload = () => resolve(String(reader.result ?? ''));
    reader.onerror = () => reject(reader.error ?? new Error('Failed to read image file.'));
    reader.readAsDataURL(blob);
  });
}

function readImageSize(src: string): Promise<{ width: number; height: number }> {
  if (typeof Image === 'undefined') {
    return Promise.resolve({ width: 0, height: 0 });
  }

  return new Promise((resolve) => {
    const img = new Image();

    img.onload = () => resolve({ width: img.naturalWidth, height: img.naturalHeight });
    img.onerror = () => resolve({ width: 0, height: 0 });
    img.src = src;
  });
}

export function setImageRotation(

export function rotateImageBy(

}

export function setImageOpacity(
  image: WriterImageModel,
  opacity: number,
): WriterImageModel {
  return {
    ...image,
    opacity: Math.max(0, Math.min(1, opacity)),
  };
}

export function setImageBorder(
  image: WriterImageModel,
  input: {
    borderColor?: string;
    borderWidth?: number;
  },
): WriterImageModel {
  return {
    ...image,
    borderColor: input.borderColor ?? image.borderColor,
    borderWidth:
      typeof input.borderWidth === 'number'
        ? Math.max(0, input.borderWidth)
        : image.borderWidth,
  };
}

export function setImageAlt(
  image: WriterImageModel,
  alt: string,
): WriterImageModel {
  return {
    ...image,
    alt,
  };
}

export function setImageCrop(
  image: WriterImageModel,
  crop: WriterImageCrop | undefined,
): WriterImageModel {
  if (!crop) {
    return {
      ...image,
      crop: undefined,
    };
  }

  return {
    ...image,
    crop: normalizeCrop(crop),
  };
}

export function normalizeCrop(crop: WriterImageCrop): WriterImageCrop {
  const x = clamp01(crop.x);
  const y = clamp01(crop.y);
  const width = clamp01(crop.width);
  const height = clamp01(crop.height);

  return {
    x,
    y,
    width: Math.min(width, 1 - x),
    height: Math.min(height, 1 - y),
  };
}

export function normalizeRotation(rotation: number): number {
  if (!Number.isFinite(rotation)) return 0;

  return ((rotation % 360) + 360) % 360;
}

function clamp01(value: number): number {
  if (!Number.isFinite(value)) return 0;
  return Math.max(0, Math.min(1, value));
}
